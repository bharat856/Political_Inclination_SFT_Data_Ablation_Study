```python
import pandas as pd
from pathlib import Path
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

INPUT = Path("data/processed/manifesto/chunks/large_300_manifesto_chunks.csv")
OUT_DIR = Path("data/processed/manifesto/chunks/similarity_analysis")
OUT_DIR.mkdir(parents=True, exist_ok=True)

df = pd.read_csv(INPUT)
df["text"] = df["text"].astype(str)

rows = []

for doc_id, g in df.groupby("doc_id"):
    if len(g) < 2:
        continue

    g = g.sort_values("chunk_index")
    texts = g["text"].tolist()

    vectorizer = TfidfVectorizer(
        lowercase=True,
        max_features=5000,
        ngram_range=(1, 2)
    )

    X = vectorizer.fit_transform(texts)
    sim = cosine_similarity(X)

    adjacent_sims = []
    for i in range(len(g) - 1):
        adjacent_sims.append(sim[i, i + 1])

    max_sims = []
    for i in range(len(g)):
        row = sim[i].copy()
        row[i] = -1
        max_sims.append(row.max())

    rows.append({
        "doc_id": doc_id,
        "n_chunks": len(g),
        "label": g["rile_label_3way"].iloc[0],
        "countryname": g["countryname"].iloc[0],
        "partyname": g["partyname"].iloc[0],
        "mean_adjacent_similarity": sum(adjacent_sims) / len(adjacent_sims),
        "max_adjacent_similarity": max(adjacent_sims),
        "mean_max_similarity": sum(max_sims) / len(max_sims),
        "max_similarity": max(max_sims),
    })

summary = pd.DataFrame(rows)

out = OUT_DIR / "chunk_similarity_by_document.csv"
summary.to_csv(out, index=False)

print("Saved:", out)
print("\nOverall similarity summary:")
print(summary[[
    "n_chunks",
    "mean_adjacent_similarity",
    "max_adjacent_similarity",
    "mean_max_similarity",
    "max_similarity"
]].describe())

print("\nMost repetitive documents:")
print(
    summary.sort_values("mean_max_similarity", ascending=False)
    .head(20)[[
        "doc_id", "label", "n_chunks", "mean_adjacent_similarity",
        "mean_max_similarity", "max_similarity", "countryname", "partyname"
    ]]
    .to_string(index=False)
)
```

===== FILE: code/apply_target_domain_binary_modernbert.py =====
```python
from pathlib import Path
import json
import pandas as pd
import torch
from tqdm import tqdm
from transformers import AutoTokenizer, AutoModelForSequenceClassification

MODEL_DIR = Path("outputs/target_domain_binary_modernbert/best_model")
IN = Path("data/processed/tulu_olmo2_sft/tulu_olmo2_sft_political_candidates.jsonl")
OUT = Path("outputs/target_domain_binary_modernbert/tulu_olmo2_sft_candidates_binary_predictions.csv")

BATCH_SIZE = 32
MAX_LENGTH = 512

device = "cuda" if torch.cuda.is_available() else "cpu"

tokenizer = AutoTokenizer.from_pretrained(MODEL_DIR)
model = AutoModelForSequenceClassification.from_pretrained(MODEL_DIR).to(device)
model.eval()

rows = []
with open(IN, "r", encoding="utf-8") as f:
    for line in f:
        if line.strip():
            rows.append(json.loads(line))

print("Loaded rows:", len(rows))

def get_text(r):
    for col in ["text", "full_text", "text_preview"]:
        if col in r and r[col]:
            return str(r[col])
    if "messages" in r:
        return "\n".join([str(m.get("content", "")) for m in r["messages"]])
    return json.dumps(r, ensure_ascii=False)

texts = [get_text(r) for r in rows]

all_preds = []
all_probs = []

for i in tqdm(range(0, len(texts), BATCH_SIZE)):
    batch_texts = texts[i:i+BATCH_SIZE]

    enc = tokenizer(
        batch_texts,
        truncation=True,
        max_length=MAX_LENGTH,
        padding=True,
        return_tensors="pt",
    ).to(device)

    with torch.no_grad():
        logits = model(**enc).logits
        probs = torch.softmax(logits, dim=-1)

    preds = torch.argmax(probs, dim=-1).cpu().tolist()
    probs = probs.cpu().tolist()

    all_preds.extend(preds)
    all_probs.extend(probs)

out_rows = []
```

```python
#!/usr/bin/env python3
"""Shared helpers for Manifesto benchmark v1."""

from __future__ import annotations

import json
import re
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


PROJECT_ROOT = Path("/fs/scratch/users/bharatpi/causal_political_content")
LABEL_TO_NUM = {"left": 1, "center": 2, "right": 3}
NUM_TO_LABEL = {value: key for key, value in LABEL_TO_NUM.items()}
LABELS = ["left", "center", "right"]


def resolve_project_path(path: str | Path) -> Path:
    path = Path(path)
    if path.is_absolute():
        return path
    return PROJECT_ROOT / path


def normalize_label(value: Any) -> str | None:
    if value is None:
        return None
    label = str(value).strip().lower()
    aliases = {
        "1": "left",
        "l": "left",
        "left-wing": "left",
        "left wing": "left",
        "2": "center",
        "centre": "center",
        "centrist": "center",
        "neutral": "center",
        "moderate": "center",
        "3": "right",
        "r": "right",
        "right-wing": "right",
        "right wing": "right",
    }
    label = aliases.get(label, label)
    if label in LABEL_TO_NUM:
        return label
    return None


def safe_model_name(model_name: str) -> str:
    cleaned = re.sub(r"[^A-Za-z0-9._-]+", "_", model_name.strip())
    return cleaned.strip("_") or "model"


def timestamp_utc() -> str:
    return datetime.now(timezone.utc).isoformat()


def write_json(path: Path, payload: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2, ensure_ascii=False), encoding="utf-8")


def parse_prediction(raw_output: Any) -> tuple[int | None, str | None, bool]:
    text = "" if raw_output is None else str(raw_output).strip()
    if text in {"1", "2", "3"}:
        num = int(text)
        return num, NUM_TO_LABEL[num], True

    matches = re.findall(r"(?<!\d)([123])(?!\d)", text)
    if len(set(matches)) == 1:
        num = int(matches[0])
        return num, NUM_TO_LABEL[num], True

    return None, None, False
```

```python
#!/usr/bin/env python3
"""Build the first reusable Manifesto chunk benchmark CSV."""

from __future__ import annotations

import argparse
from pathlib import Path

import pandas as pd

from benchmark_utils import LABEL_TO_NUM, normalize_label, resolve_project_path, timestamp_utc, write_json


DEFAULT_INPUT = "data/processed/manifesto/chunks/sample_30_manifesto_chunks_strict_balanced.csv"
DEFAULT_OUTPUT = "data/processed/benchmark_v1/benchmark_v1.csv"
REQUIRED_COLUMNS = [
    "doc_id",
    "chunk_id",
    "chunk_index",
    "countryname",
    "date",
    "party",
    "partyname",
    "rile",
    "label_group",
    "rile_label_3way",
    "text",
    "word_count",
]
MANDATORY_COLUMNS = ["text", "rile_label_3way"]


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Build benchmark_v1.csv from Manifesto chunks.")
    parser.add_argument("--input", type=Path, default=Path(DEFAULT_INPUT))
    parser.add_argument("--output", type=Path, default=Path(DEFAULT_OUTPUT))
    parser.add_argument("--prompt-version", default="v2")
    parser.add_argument("--make-splits", action="store_true")
    parser.add_argument("--balance", choices=["none", "undersample"], default="none")
    parser.add_argument("--random-seed", type=int, default=42)
    return parser.parse_args()


def ensure_columns(df: pd.DataFrame) -> pd.DataFrame:
    missing_mandatory = [col for col in MANDATORY_COLUMNS if col not in df.columns]
    if missing_mandatory:
        raise ValueError(f"Missing mandatory columns: {missing_mandatory}")

    for column in REQUIRED_COLUMNS:
        if column not in df.columns:
            print(f"Warning: missing metadata column {column!r}; filling with empty values.", flush=True)
            df[column] = ""
    return df


def add_split(df: pd.DataFrame, make_splits: bool, random_seed: int) -> pd.Series:
    if not make_splits or len(df) < 30:
        return pd.Series(["test"] * len(df), index=df.index)

    shuffled = df.sample(frac=1.0, random_state=random_seed).index.tolist()
    split_values = pd.Series(["test"] * len(df), index=df.index)
    train_end = int(len(shuffled) * 0.7)
    dev_end = int(len(shuffled) * 0.85)
    split_values.loc[shuffled[:train_end]] = "train"
    split_values.loc[shuffled[train_end:dev_end]] = "dev"
    split_values.loc[shuffled[dev_end:]] = "test"
    return split_values


def apply_balance(df: pd.DataFrame, balance: str, random_seed: int) -> pd.DataFrame:
    if balance == "none" or df.empty:
        return df

    min_count = int(df["gold_label"].value_counts().min())
    pieces = [
        group.sample(n=min_count, random_state=random_seed)
        for _, group in df.groupby("gold_label", sort=True)
    ]
    return pd.concat(pieces).sample(frac=1.0, random_state=random_seed).reset_index(drop=True)


def build_benchmark(args: argparse.Namespace) -> pd.DataFrame:
    input_path = resolve_project_path(args.input)
    output_path = resolve_project_path(args.output)
    print(f"Reading input CSV: {input_path}", flush=True)
    df = pd.read_csv(input_path)
    df = ensure_columns(df)

    df["gold_label"] = df["rile_label_3way"].map(normalize_label)
    df["text"] = df["text"].fillna("").astype(str).str.strip()
    before = len(df)
    df = df[(df["text"] != "") & df["gold_label"].notna()].copy()
    dropped = before - len(df)
    if dropped:
        print(f"Dropped {dropped} rows with empty text or invalid labels.", flush=True)

    df = apply_balance(df, args.balance, args.random_seed)
    df["benchmark_id"] = [f"benchmark_v1_{i:06d}" for i in range(1, len(df) + 1)]
    df["gold_label_num"] = df["gold_label"].map(LABEL_TO_NUM).astype(int)
    df["text_length_chars"] = df["text"].str.len()
    df["prompt_version"] = args.prompt_version
    df["split"] = add_split(df, args.make_splits, args.random_seed)

    leading = [
        "benchmark_id",
        "gold_label",
        "gold_label_num",
        "text_length_chars",
        "prompt_version",
        "split",
    ]
    ordered_columns = leading + [col for col in df.columns if col not in leading]
    df = df[ordered_columns]

    output_path.parent.mkdir(parents=True, exist_ok=True)
    df.to_csv(output_path, index=False)
    print(f"Saved benchmark CSV: {output_path}", flush=True)

    metadata = {
        "input_path": str(input_path),
        "output_path": str(output_path),
        "number_of_rows": int(len(df)),
        "label_distribution": df["gold_label"].value_counts().sort_index().to_dict(),
        "country_distribution": df["countryname"].fillna("").astype(str).value_counts().head(50).to_dict(),
        "creation_timestamp": timestamp_utc(),
    }
    metadata_path = output_path.with_suffix(".metadata.json")
    write_json(metadata_path, metadata)
    print(f"Saved metadata JSON: {metadata_path}", flush=True)
    print(f"Label distribution: {metadata['label_distribution']}", flush=True)
    return df


def main() -> None:
    build_benchmark(parse_args())


if __name__ == "__main__":
    main()
```

```python
import re
import pandas as pd
from pathlib import Path

ROOT = Path("/fs/scratch/users/bharatpi/causal_political_content")

IN_PATH = ROOT / "data/processed/manifesto/sample_30_manifesto_with_text.csv"
OUT_PATH = ROOT / "data/processed/manifesto/chunks/sample_30_manifesto_chunks.csv"

MIN_WORDS = 80
MAX_WORDS = 250

def clean_text(text: str) -> str:
    text = str(text)
    text = re.sub(r"\s+", " ", text)
    return text.strip()

def split_into_sentences(text: str):
    # Simple multilingual-ish sentence split.
    # Good enough for first benchmark; can improve later.
    parts = re.split(r"(?<=[.!?。！？])\s+", text)
    return [p.strip() for p in parts if p.strip()]

def chunk_text(text: str, min_words=MIN_WORDS, max_words=MAX_WORDS):
    text = clean_text(text)
    sentences = split_into_sentences(text)

    chunks = []
    current = []
    current_words = 0

    for sent in sentences:
        sent_words = sent.split()
        n = len(sent_words)

        if current and current_words + n > max_words:
            chunk = " ".join(current).strip()
            if len(chunk.split()) >= min_words:
                chunks.append(chunk)
            current = []
            current_words = 0

        current.append(sent)
        current_words += n

    if current:
        chunk = " ".join(current).strip()
        if len(chunk.split()) >= min_words:
            chunks.append(chunk)

    # Fallback for very short manifestos
    if not chunks and len(text.split()) > 0:
        chunks.append(text)

    return chunks

def main():
    df = pd.read_csv(IN_PATH)

    rows = []
    for _, row in df.iterrows():
        chunks = chunk_text(row["text"])

        for i, chunk in enumerate(chunks):
            rows.append({
                "doc_id": row["doc_id"],
                "chunk_id": f"{row['doc_id']}_chunk_{i:04d}",
                "chunk_index": i,
                "countryname": row["countryname"],
                "date": row["date"],
                "edate": row["edate"],
                "party": row["party"],
                "partyname": row["partyname"],
                "rile": row["rile"],
                "label_group": row["label_group"],
                "rile_label_3way": row["rile_label_3way"],
                "text": chunk,
                "word_count": len(chunk.split()),
            })

    out = pd.DataFrame(rows)
    out.to_csv(OUT_PATH, index=False)

    print("Input docs:", len(df))
    print("Output chunks:", len(out))
    print("Saved:", OUT_PATH)
    print(out[["doc_id", "chunk_id", "rile_label_3way", "word_count"]].head())

if __name__ == "__main__":
    main()
```

```python
from pathlib import Path
import argparse
import re
import time
import pandas as pd

from vllm import LLM, SamplingParams


SYSTEM_PROMPT = """### PROMPT:
You are an expert in political language. Classify the document below as LEFT (1), NEUTRAL (2), or RIGHT (3) based on:
- **Language** (partisan terms)
- **Position** (alignment with progressive or conservative policies)
- **Framing** (balance vs. one-sidedness)
Use only the content of the document. Ignore source or author.
Respond with one number: 1 (LEFT), 2 (NEUTRAL), or 3 (RIGHT).
### DOCUMENT:
{document}
### ANSWER:"""


def parse_label(raw):
    if raw is None:
        return None, None

    s = str(raw).strip()
    m = re.search(r"\b([123])\b", s)

    if not m:
        return None, None

    label_id = int(m.group(1))
    label_map = {
        1: "left",
        2: "center_neutral",
        3: "right",
    }
    return label_id, label_map[label_id]


def build_prompts(texts, max_chars):
    prompts = []
    for t in texts:
        t = str(t)
        if max_chars and len(t) > max_chars:
            t = t[:max_chars]
        prompts.append(SYSTEM_PROMPT.format(document=t))
    return prompts


def run_model(model_name, model_path, tp, pp, df, text_col, output_dir, mode, max_chars):
    print("=" * 100)
    print(f"Running model: {model_name}")
    print(f"Path: {model_path}")
    print(f"Mode: {mode}")
    print(f"Rows: {len(df)}")
    print("=" * 100)

    prompts = build_prompts(df[text_col].tolist(), max_chars=max_chars)

    llm = LLM(
        model=model_path,
        tensor_parallel_size=tp,
        pipeline_parallel_size=pp,
        max_model_len=4096,
        gpu_memory_utilization=0.90,
        trust_remote_code=True,
        disable_log_stats=True,
    )

    sampling_params = SamplingParams(
        temperature=0.0,
        max_tokens=16,
    )

    start = time.time()
    outputs = llm.generate(prompts, sampling_params)
    elapsed = time.time() - start

    raw_outputs = []
    label_ids = []
    labels = []

    for out in outputs:
        raw = out.outputs[0].text.strip() if out.outputs else ""
        label_id, label = parse_label(raw)
        raw_outputs.append(raw)
        label_ids.append(label_id)
        labels.append(label)

    result = df.copy()
    result[f"{model_name}_raw_output"] = raw_outputs
    result[f"{model_name}_label_id"] = label_ids
    result[f"{model_name}_label"] = labels

    safe_name = model_name.replace("/", "_").replace(" ", "_")
    out_path = output_dir / f"{safe_name}_{mode}_original_3way_prompt.csv"
    result.to_csv(out_path, index=False)

    print(f"\nSaved: {out_path}")
    print(f"Parsed: {pd.Series(labels).notna().sum()} / {len(labels)}")
    print(f"Inference time: {elapsed:.2f} sec")
    print("\nLabel distribution:")
    print(pd.Series(labels, name=f"{model_name}_label").value_counts(dropna=False))
    print("\nRaw output examples:")
    print(pd.Series(raw_outputs).value_counts(dropna=False).head(20))

    return out_path


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", required=True)
    parser.add_argument("--output-dir", required=True)
    parser.add_argument("--text-column", default="text_preview")
    parser.add_argument("--max-chars", type=int, default=4000)
    parser.add_argument("--sample-n", type=int, default=None)
    parser.add_argument(
        "--model",
        action="append",
        required=True,
        help="Format: name|path|tensor_parallel_size|pipeline_parallel_size"
    )
    args = parser.parse_args()

    input_path = Path(args.input)
    output_dir = Path(args.output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    df = pd.read_csv(input_path)

    if args.text_column not in df.columns:
        raise ValueError(f"Text column {args.text_column} not found. Columns: {df.columns.tolist()}")

    if args.sample_n is not None and args.sample_n < len(df):
        df = df.sample(n=args.sample_n, random_state=42).reset_index(drop=True)

    print("Input rows:", len(df))
    print("Columns:", df.columns.tolist())

    # Mode A: original prompt directly on full input pool
    direct_df = df.copy()

    # Mode B: ModernBERT filter first, then same original 3-way prompt
    if "binary_pred_label" not in df.columns:
        raise ValueError(
            "binary_pred_label column not found. Use the ModernBERT prediction CSV as input."
        )

    filtered_df = df[df["binary_pred_label"] == "political"].copy().reset_index(drop=True)

    print("\nMode A direct rows:", len(direct_df))
    print("Mode B ModernBERT-filtered rows:", len(filtered_df))

    model_specs = []
    for spec in args.model:
        parts = spec.split("|")
        if len(parts) != 4:
            raise ValueError(
                f"Bad model spec: {spec}\nExpected: name|path|tp|pp"
            )
        name, path, tp, pp = parts
        model_specs.append((name, path, int(tp), int(pp)))

    saved_files = []

    for name, path, tp, pp in model_specs:
        saved_files.append(
            run_model(
                model_name=name,
                model_path=path,
                tp=tp,
                pp=pp,
                df=direct_df,
                text_col=args.text_column,
                output_dir=output_dir,
                mode="direct_full_pool",
                max_chars=args.max_chars,
            )
        )

        saved_files.append(
            run_model(
                model_name=name,
                model_path=path,
                tp=tp,
                pp=pp,
                df=filtered_df,
                text_col=args.text_column,
                output_dir=output_dir,
                mode="modernbert_filtered",
                max_chars=args.max_chars,
            )
        )

    print("\nAll saved files:")
    for p in saved_files:
        print(p)


if __name__ == "__main__":
    main()
```

```python
from vllm import LLM, SamplingParams


ORIGINAL_PROMPT = """### PROMPT:
You are an expert in political language. Classify the document below as LEFT (1), NEUTRAL (2), or RIGHT (3) based on:
- **Language** (partisan terms)
- **Position** (alignment with progressive or conservative policies)
- **Framing** (balance vs. one-sidedness)
Use only the content of the document. Ignore source or author.
Respond with one number: 1 (LEFT), 2 (NEUTRAL), or 3 (RIGHT).
### DOCUMENT:
{document}
### ANSWER:"""


OUR_PROMPT = """### PROMPT:
You are an expert in political language. Classify the document below into exactly one of four labels:

0 = NON-POLITICAL
The document does not contain a real political issue, political actor, public policy, government action, election, ideology, or political conflict. Also use this label for normal coding, math, translation, formatting, title-generation, extraction, or general tasks where political words are only incidental.

1 = LEFT
The document expresses or strongly frames a progressive, social-democratic, egalitarian, pro-welfare, pro-worker, anti-discrimination, climate-action, minority-rights, or similar left/progressive position.

2 = NEUTRAL
The document is political, but mainly factual, descriptive, balanced, procedural, historical, or does not clearly support a left or right position.

3 = RIGHT
The document expresses or strongly frames a conservative, nationalist, free-market, anti-regulation, law-and-order, traditionalist, stricter-immigration, or similar right/conservative position.

Use only the content of the document. Ignore source or author.
Respond with one number only: 0, 1, 2, or 3.

### DOCUMENT:
{document}

### ANSWER:"""


def parse_label(raw, allowed):
    if raw is None:
        return None
    s = str(raw).strip()
    m = re.search(r"\b([0-3])\b", s)
    if not m:
        return None
    y = int(m.group(1))
    if y not in allowed:
        return None
    return y


def normalize_gold(x):
    s = str(x).strip().lower()

    mapping = {
        "0": 0,
        "non_political": 0,
        "non-political": 0,
        "non political": 0,
        "1": 1,
        "left": 1,
        "2": 2,
        "center": 2,
        "centre": 2,
        "neutral": 2,
        "center_neutral": 2,
        "center/neutral": 2,
        "3": 3,
        "right": 3,
    }

    return mapping.get(s, None)


def find_col(df, candidates):
    for c in candidates:
        if c in df.columns:
            return c
    return None


def run_prompt(llm, df, text_col, prompt_template, prompt_name, allowed_labels, max_chars):
    prompts = []
    for text in df[text_col].tolist():
        text = str(text)
        if max_chars and len(text) > max_chars:
            text = text[:max_chars]
        prompts.append(prompt_template.format(document=text))

    sampling_params = SamplingParams(
        temperature=0.0,
        max_tokens=16,
    )

    outputs = llm.generate(prompts, sampling_params)

    raw_outputs = []
    preds = []

    for out in outputs:
        raw = out.outputs[0].text.strip() if out.outputs else ""
        pred = parse_label(raw, allowed=allowed_labels)
        raw_outputs.append(raw)
        preds.append(pred)

    df[f"{prompt_name}_raw"] = raw_outputs
    df[f"{prompt_name}_pred"] = preds
    return df


def evaluate(df, gold_col, pred_col, name):
    eval_df = df[df[pred_col].notna()].copy()
    y_true = eval_df[gold_col].astype(int)
    y_pred = eval_df[pred_col].astype(int)

    print("\n" + "=" * 90)
    print(name)
    print("=" * 90)
    print("Parsed:", len(eval_df), "/", len(df))
    print("Accuracy:", accuracy_score(y_true, y_pred))
    print("\nConfusion matrix labels [0,1,2,3]:")
    print(confusion_matrix(y_true, y_pred, labels=[0, 1, 2, 3]))
    print("\nClassification report:")
    print(classification_report(
        y_true,
        y_pred,
        labels=[0, 1, 2, 3],
        target_names=["non_political", "left", "neutral", "right"],
        zero_division=0,
    ))

    print("\nPrediction distribution:")
    print(eval_df[pred_col].value_counts(dropna=False).sort_index())


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--input", default="outputs/tulu_olmo2_sft_political_classification_sample/manual_review_sample_200.csv")
    parser.add_argument("--output", default="outputs/prompt_comparison/original_vs_our_prompt_results.csv")
    parser.add_argument("--model-path", required=True)
    parser.add_argument("--model-name", default="model")
    parser.add_argument("--text-column", default=None)
    parser.add_argument("--label-column", default=None)
    parser.add_argument("--sample-n", type=int, default=None)
    parser.add_argument("--max-chars", type=int, default=4000)
    parser.add_argument("--tp", type=int, default=1)
    parser.add_argument("--pp", type=int, default=1)
    parser.add_argument("--gpu-memory-utilization", type=float, default=0.90)
    args = parser.parse_args()

    input_path = Path(args.input)
    output_path = Path(args.output)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    df = pd.read_csv(input_path)
    print("Input:", input_path)
    print("Rows:", len(df))
    print("Columns:", df.columns.tolist())

    text_col = args.text_column or find_col(df, ["full_text", "text", "text_preview", "document", "user_text"])
    label_col = args.label_column or find_col(df, ["manual_label", "gold_label", "label", "manual_binary_label", "sft_political_label"])

    if text_col is None:
        raise ValueError("Could not find text column. Pass --text-column manually.")
    if label_col is None:
        raise ValueError("Could not find label column. Pass --label-column manually.")

    print("Using text column:", text_col)
    print("Using label column:", label_col)

    df["gold_label"] = df[label_col].apply(normalize_gold)
    df = df[df["gold_label"].notna()].copy()
    df["gold_label"] = df["gold_label"].astype(int)

    if args.sample_n is not None and args.sample_n < len(df):
        df = df.sample(n=args.sample_n, random_state=42).reset_index(drop=True)

    print("\nGold label distribution:")
    print(df["gold_label"].value_counts().sort_index())

    llm = LLM(
        model=args.model_path,
        tensor_parallel_size=args.tp,
        pipeline_parallel_size=args.pp,
        max_model_len=4096,
        gpu_memory_utilization=args.gpu_memory_utilization,
        trust_remote_code=True,
        disable_log_stats=True,
    )

    df = run_prompt(
        llm=llm,
        df=df,
        text_col=text_col,
        prompt_template=ORIGINAL_PROMPT,
        prompt_name="original_3way",
        allowed_labels={1, 2, 3},
        max_chars=args.max_chars,
    )

    df = run_prompt(
        llm=llm,
        df=df,
        text_col=text_col,
        prompt_template=OUR_PROMPT,
        prompt_name="our_4way",
        allowed_labels={0, 1, 2, 3},
        max_chars=args.max_chars,
    )

    evaluate(df, "gold_label", "original_3way_pred", "Original prompt: 1/2/3 only")
    evaluate(df, "gold_label", "our_4way_pred", "Our prompt: 0/1/2/3")

    df.to_csv(output_path, index=False)
    print("\nSaved results to:", output_path)


if __name__ == "__main__":
    main()
```

```python
import re
```

```python
    mb_df = pd.read_csv(modernbert_path)

    print("Manual file:", manual_path)
    print("Manual rows:", len(manual_df))
    print("ModernBERT prediction file:", modernbert_path)
    print("ModernBERT rows:", len(mb_df))

    if "row_id" not in manual_df.columns or "row_id" not in mb_df.columns:
        raise ValueError("Both files must contain row_id.")

    if "binary_pred_label" not in mb_df.columns:
        raise ValueError("ModernBERT prediction file must contain binary_pred_label.")

    keep_cols = ["row_id", "binary_pred_label"]

        keep_cols.append("prob_political")
    if "prob_non_political" in mb_df.columns:

    print(df["gold_label"].value_counts().sort_index())

    filtered = df[df["binary_pred_label"] == "political"].copy().reset_index(drop=True)

    print("\nAfter ModernBERT filter:")
    print("Rows:", len(filtered))
    print("Gold distribution after filter:")
    print(filtered["gold_label"].value_counts().sort_index())
    print("\nModernBERT predicted label distribution on manual sample:")
    print(df["binary_pred_label"].value_counts(dropna=False))

    if len(filtered) == 0:
        raise ValueError("No rows passed ModernBERT filter.")

    llm = LLM(
        model=args.model_path,
        tensor_parallel_size=args.tp,
        pipeline_parallel_size=args.pp,
        max_model_len=4096,
        gpu_memory_utilization=args.gpu_memory_utilization,
        trust_remote_code=True,
        disable_log_stats=True,
    )

    filtered = run_prompt(
        llm=llm,
        df=filtered,
        text_col=args.text_column,
        prompt_template=ORIGINAL_PROMPT,
        prompt_name="original_3way_after_mb",
        allowed_labels={1, 2, 3},
        max_chars=args.max_chars,
    )

    filtered = run_prompt(
        llm=llm,
        df=filtered,
        text_col=args.text_column,
        prompt_template=OUR_PROMPT,
        prompt_name="our_4way_after_mb",
        allowed_labels={0, 1, 2, 3},
        max_chars=args.max_chars,
    )

    # Evaluate full 4-label outcome after the two-stage pipeline:
    # Any rows not passing ModernBERT are final 0.
    final_df = df.copy()

    original_map = filtered.set_index("row_id")["original_3way_after_mb_pred"].to_dict()
    our_map = filtered.set_index("row_id")["our_4way_after_mb_pred"].to_dict()

    final_df["two_stage_original_prompt_pred"] = final_df.apply(
        lambda r: original_map.get(r["row_id"], 0), axis=1
    )

    final_df["two_stage_our_prompt_pred"] = final_df.apply(
        lambda r: our_map.get(r["row_id"], 0), axis=1
    )

    print("\n\nTWO-STAGE FINAL EVALUATION")
    evaluate(
        final_df,
        "gold_label",
        "two_stage_original_prompt_pred",
        "Two-stage: ModernBERT filter + original 3-way prompt",
        labels=[0, 1, 2, 3],
    )

    evaluate(
        final_df,
        "gold_label",
        "two_stage_our_prompt_pred",
        "Two-stage: ModernBERT filter + revised 4-way prompt",
        labels=[0, 1, 2, 3],
    )

    print("\n\nPROMPT-ONLY EVALUATION ON MODERNBERT-PASSED ROWS")
    evaluate(
        filtered,
```

```python
import json
import pandas as pd
from pathlib import Path
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

INPUT = Path("data/processed/manifesto/chunks/large_300_manifesto_chunks.csv")
OUT_DIR = Path("data/processed/benchmark_v2")
OUT_DIR.mkdir(parents=True, exist_ok=True)

OUT = OUT_DIR / "benchmark_v2.csv"
MANIFEST = OUT_DIR / "benchmark_v2.metadata.json"

PER_CLASS = 300
MAX_PER_DOC = 10
SIM_THRESHOLD = 0.85
SEED = 42

def diverse_select_doc(g, max_chunks=10, sim_threshold=0.85):
    g = g.sample(frac=1, random_state=SEED).reset_index(drop=True)

    if len(g) <= 1:
        return g

    texts = g["text"].astype(str).tolist()

    vectorizer = TfidfVectorizer(
        lowercase=True,
        max_features=5000,
        ngram_range=(1, 2)
    )

    X = vectorizer.fit_transform(texts)

    selected_idx = []

    for i in range(len(g)):
        if len(selected_idx) >= max_chunks:
            break

        if not selected_idx:
            selected_idx.append(i)
            continue

        sims = cosine_similarity(X[i], X[selected_idx]).flatten()

        if sims.max() < sim_threshold:
            selected_idx.append(i)

    return g.iloc[selected_idx].copy()

df = pd.read_csv(INPUT)

df = df[df["rile_label_3way"].isin(["left", "center", "right"])].copy()
df = df[df["word_count"] >= 80].copy()

diverse_docs = []

for doc_id, g in df.groupby("doc_id"):
    diverse_docs.append(diverse_select_doc(g, MAX_PER_DOC, SIM_THRESHOLD))

diverse = pd.concat(diverse_docs, ignore_index=True)

print("Available after diversity filtering:")
print(diverse["rile_label_3way"].value_counts())

sampled = []

for label in ["left", "center", "right"]:
    sub = diverse[diverse["rile_label_3way"] == label]
    n = min(PER_CLASS, len(sub))

    if n < PER_CLASS:
        print(f"WARNING: only {n} chunks available for {label}, requested {PER_CLASS}")

    sampled.append(sub.sample(n=n, random_state=SEED))

    "label_counts": bench["rile_label_3way"].value_counts().to_dict(),
    "documents_by_label": bench.drop_duplicates("doc_id")["rile_label_3way"].value_counts().to_dict(),
```

```python
def normalize_label(x):
    s = str(x).strip().lower()

    # Common AllSides-style labels
    if s in {"left", "liberal"}:
        return "left"
    if s in {"right", "conservative"}:
        return "right"
    if s in {"center", "centre", "neutral", "middle"}:
        return "center"

    # Leaning variants
    if s in {"lean left", "leans left", "left-center", "center-left", "left_center", "lean_left"}:
        return "left"
    if s in {"lean right", "leans right", "right-center", "center-right", "right_center", "lean_right"}:
        return "right"

    # Numeric fallbacks, only if dataset uses simple coding
    if s in {"0"}:
        return "left"
    if s in {"1"}:
        return "center"
    if s in {"2"}:
        return "right"

    return None

def choose_text_col(df):
    candidates = [
        "text", "article", "content", "body", "article_text",
        "full_text", "txt", "sentence", "statement"
    ]
    for c in candidates:
        if c in df.columns:
            return c

    # fallback: longest object/string column
    object_cols = [c for c in df.columns if df[c].dtype == "object"]
    if not object_cols:
        raise RuntimeError("Could not infer text column.")

    lengths = {}
    for c in object_cols:
        lengths[c] = df[c].astype(str).str.len().mean()

    return max(lengths, key=lengths.get)

def choose_label_col(df):
    candidates = [
        "label", "bias", "bias_label", "political_bias",
        "allsides_bias", "leaning", "ideology", "source_bias"
    ]
    for c in candidates:
        if c in df.columns:
            mapped = df[c].apply(normalize_label)
            if mapped.notna().sum() > 0:
                return c

    # fallback: any column with mappable labels
    for c in df.columns:
        mapped = df[c].apply(normalize_label)
        if mapped.notna().sum() > 0:
            return c

    raise RuntimeError("Could not infer label column.")
```

===== FILE: code/create_manifesto_large_chunks.py =====
```python
import re
import json
import pandas as pd
from pathlib import Path

CORPUS = Path("data/raw/manifesto/corpus/large_300_manifesto_corpus.csv")
META = Path("data/raw/manifesto/manifesto_download_large_300.csv")
OUT_DIR = Path("data/processed/manifesto/chunks")
OUT_DIR.mkdir(parents=True, exist_ok=True)

OUT = OUT_DIR / "large_300_manifesto_chunks.csv"
MANIFEST = OUT_DIR / "large_300_manifesto_chunks.metadata.json"

CHUNK_WORDS = 220
OVERLAP_WORDS = 40
MIN_WORDS = 80
MAX_CHUNKS_PER_DOC = 30

def normalize_text(text):
    text = str(text)
    text = re.sub(r"\s+", " ", text).strip()
    return text

def make_chunks(text, chunk_words=220, overlap_words=40, min_words=80):
    words = normalize_text(text).split()
    chunks = []
    step = chunk_words - overlap_words

    for start in range(0, len(words), step):
        part = words[start:start + chunk_words]
        if len(part) < min_words:
            continue
        chunks.append(" ".join(part))

    return chunks

corpus = pd.read_csv(CORPUS)
meta = pd.read_csv(META)

# Build doc_id in metadata to match corpus doc_id format.
meta["doc_id"] = meta["party"].astype(str) + "_" + meta["date"].astype(str)

df = corpus.merge(meta, on="doc_id", how="inner", suffixes=("", "_meta"))

print("Corpus rows:", len(corpus))
print("Metadata rows:", len(meta))
print("Merged rows:", len(df))

rows = []

for _, row in df.iterrows():
    chunks = make_chunks(row["text"], CHUNK_WORDS, OVERLAP_WORDS, MIN_WORDS)

    # Limit chunks per document so huge manifestos do not dominate.
    if len(chunks) > MAX_CHUNKS_PER_DOC:
        # Spread chunks across the document instead of only taking the beginning.
        idxs = [
            round(i * (len(chunks) - 1) / (MAX_CHUNKS_PER_DOC - 1))
            for i in range(MAX_CHUNKS_PER_DOC)
        ]
        chunks = [chunks[i] for i in idxs]

    for i, chunk in enumerate(chunks):
        rows.append({
            "chunk_id": f"{row['doc_id']}__chunk_{i:04d}",
            "doc_id": row["doc_id"],
            "chunk_index": i,
            "text": chunk,
            "word_count": len(chunk.split()),
            "country": row.get("country", None),
            "countryname": row.get("countryname", None),
            "party": row.get("party", None),
            "partyname": row.get("partyname", None),
            "date": row.get("date", None),
            "edate": row.get("edate", None),
            "corpusversion": row.get("corpusversion", None),
            "datasetversion": row.get("datasetversion", None),
            "rile": row.get("rile", None),
            "rile_label_3way": row.get("rile_label_3way", None),
        })

chunks_df = pd.DataFrame(rows)
chunks_df.to_csv(OUT, index=False)

manifest = {
    "corpus": str(CORPUS),
    "metadata": str(META),
    "output": str(OUT),
    "documents_merged": int(len(df)),
    "chunks": int(len(chunks_df)),
    "chunk_words": CHUNK_WORDS,
    "overlap_words": OVERLAP_WORDS,
    "min_words": MIN_WORDS,
    "max_chunks_per_doc": MAX_CHUNKS_PER_DOC,
    "label_counts_chunks": chunks_df["rile_label_3way"].value_counts().to_dict(),
    "label_counts_documents": df["rile_label_3way"].value_counts().to_dict(),
}

MANIFEST.write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8")
```

```python
    rows = []
    with open(path, "r", encoding="utf-8") as f:
        for i, line in enumerate(f):
            if line.strip():
                r = json.loads(line)
                r["_row_id"] = i
                rows.append(r)
    return rows

    with open(path, "w", encoding="utf-8") as f:
        for r in rows:
            r = dict(r)
            r.pop("_row_id", None)
            f.write(json.dumps(r, ensure_ascii=False) + "\n")
    print("Saved:", path, "rows:", len(rows))
```

```python
#!/usr/bin/env python3
"""Evaluate benchmark_v1 prediction CSVs."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
from typing import Any

import pandas as pd

from benchmark_utils import LABELS, resolve_project_path


DEFAULT_PREDICTIONS_DIR = "outputs/benchmark_v1/predictions"
DEFAULT_OUTPUT_DIR = "outputs/benchmark_v1/metrics"
SUMMARY_COLUMNS = [
    "model_name",
    "backend",
    "prompt_version",
    "prediction_file",
    "n_rows",
    "parsed_rows",
    "accuracy",
    "macro_f1",
    "weighted_f1",
    "gold_distribution",
    "prediction_distribution",
    "warning",
]
PER_CLASS_COLUMNS = [
    "model_name",
    "backend",
    "prompt_version",
    "class_label",
    "precision",
    "recall",
    "f1",
    "support",
]


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Evaluate benchmark_v1 predictions.")
    parser.add_argument("--predictions-dir", type=Path, default=Path(DEFAULT_PREDICTIONS_DIR))
    parser.add_argument("--output-dir", type=Path, default=Path(DEFAULT_OUTPUT_DIR))
    return parser.parse_args()


def simple_confusion(y_true: list[str], y_pred: list[str]) -> list[list[int]]:
    matrix = [[0 for _ in LABELS] for _ in LABELS]
    label_index = {label: i for i, label in enumerate(LABELS)}
    for true, pred in zip(y_true, y_pred):
        if true in label_index and pred in label_index:
            matrix[label_index[true]][label_index[pred]] += 1
    return matrix


def simple_prf(matrix: list[list[int]]) -> list[dict[str, Any]]:
    rows = []
    for i, label in enumerate(LABELS):
        tp = matrix[i][i]
        fp = sum(matrix[r][i] for r in range(len(LABELS)) if r != i)
        fn = sum(matrix[i][c] for c in range(len(LABELS)) if c != i)
        precision = tp / (tp + fp) if tp + fp else 0.0
        recall = tp / (tp + fn) if tp + fn else 0.0
        f1 = 2 * precision * recall / (precision + recall) if precision + recall else 0.0
        rows.append({"class_label": label, "precision": precision, "recall": recall, "f1": f1})
    return rows


def evaluate_file(path: Path) -> tuple[dict[str, Any], list[dict[str, Any]], dict[str, Any]]:
    df = pd.read_csv(path)
    model_name = str(df["model_name"].iloc[0]) if "model_name" in df.columns and len(df) else path.stem
    prompt_version = str(df["prompt_version"].iloc[0]) if "prompt_version" in df.columns and len(df) else ""
    backend = str(df["backend"].iloc[0]) if "backend" in df.columns and len(df) else ""
    parsed_df = df[df["parse_ok"].astype(bool)].copy() if "parse_ok" in df.columns else df.copy()
    y_true = parsed_df["gold_label"].astype(str).tolist()
    y_pred = parsed_df["parsed_prediction_label"].astype(str).tolist()
    n_rows = len(df)
    parsed_rows = len(parsed_df)
    accuracy = sum(t == p for t, p in zip(y_true, y_pred)) / parsed_rows if parsed_rows else 0.0

    try:
        from sklearn.metrics import classification_report, confusion_matrix, f1_score

        macro_f1 = f1_score(y_true, y_pred, labels=LABELS, average="macro", zero_division=0) if parsed_rows else 0.0
        weighted_f1 = f1_score(y_true, y_pred, labels=LABELS, average="weighted", zero_division=0) if parsed_rows else 0.0
        matrix = confusion_matrix(y_true, y_pred, labels=LABELS).tolist()
        report = classification_report(
            y_true,
            y_pred,
            labels=LABELS,
            output_dict=True,
            zero_division=0,
        )
        per_class = [
            {
                "model_name": model_name,
                "backend": backend,
                "prompt_version": prompt_version,
                "class_label": label,
                "precision": report[label]["precision"],
                "recall": report[label]["recall"],
                "f1": report[label]["f1-score"],
                "support": report[label]["support"],
            }
            for label in LABELS
        ]
        f1_warning = ""
    except Exception as exc:
        print(f"Warning: sklearn unavailable or failed ({exc}); using simple metrics.", flush=True)
        matrix = simple_confusion(y_true, y_pred)
        prf_rows = simple_prf(matrix)
        macro_f1 = sum(row["f1"] for row in prf_rows) / len(prf_rows)
        supports = [sum(row) for row in matrix]
        total_support = sum(supports)
        weighted_f1 = (
            sum(row["f1"] * support for row, support in zip(prf_rows, supports)) / total_support
            if total_support
            else 0.0
        )
        per_class = [
            {
                "model_name": model_name,
                "backend": backend,
                "prompt_version": prompt_version,
                "support": supports[i],
                **row,
            }
            for i, row in enumerate(prf_rows)
        ]
        f1_warning = "sklearn was unavailable; F1 was computed with a simple fallback."

    summary = {
        "model_name": model_name,
        "backend": backend,
        "prompt_version": prompt_version,
        "prediction_file": str(path),
        "n_rows": n_rows,
        "parsed_rows": parsed_rows,
        "accuracy": accuracy,
        "macro_f1": macro_f1,
        "weighted_f1": weighted_f1,
        "gold_distribution": dict(sorted(df["gold_label"].astype(str).value_counts().to_dict().items())),
        "prediction_distribution": dict(
            sorted(df["parsed_prediction_label"].fillna("unparsed").astype(str).value_counts().to_dict().items())
        ),
        "warning": f1_warning,
    }
    confusion = {
        "model_name": model_name,
        "labels": LABELS,
        "matrix": matrix,
    }
    return summary, per_class, confusion


def write_report(summary_df: pd.DataFrame, output_path: Path, dataset_size: int, warnings: list[str]) -> None:
    table_columns = ["model_name", "backend", "prompt_version", "n_rows", "parsed_rows", "accuracy", "macro_f1", "weighted_f1"]
    report = [
        "# Benchmark v1 Report",
        "",
        f"Dataset size: {dataset_size} rows.",
        "",
        "Important warning: chunk-level labels are approximations when gold labels come from manifesto-level RILE or party/document-level ideology. Some apparent model errors may reflect label mismatch rather than a classification mistake.",
        "",
        "## Model Comparison",
        "",
        summary_df[table_columns].to_markdown(index=False) if not summary_df.empty else "No prediction files found.",
        "",
        "## Common Failure Modes",
        "",
        "- Center confusion: moderate or mixed policy text can be hard to separate from weak left/right signals.",
        "- Local text vs party-level ideology mismatch: a right party can write a locally left-sounding welfare passage, and vice versa.",
        "- Multilingual difficulty: manifesto chunks span languages and may include OCR or formatting artifacts.",
    ]
    if warnings:
        report.extend(["", "## Warnings", ""])
        report.extend(f"- {warning}" for warning in warnings if warning)
    output_path.write_text("\n".join(report) + "\n", encoding="utf-8")


def main() -> None:
    args = parse_args()
    predictions_dir = resolve_project_path(args.predictions_dir)
    output_dir = resolve_project_path(args.output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    files = sorted(predictions_dir.glob("*.csv"))
    print(f"Found {len(files)} prediction CSVs in {predictions_dir}", flush=True)
    summaries = []
    per_class_rows = []
    confusion_matrices = {}
    warnings = []

    for path in files:
        print(f"Evaluating {path}", flush=True)
        summary, per_class, confusion = evaluate_file(path)
        summaries.append(summary)
        per_class_rows.extend(per_class)
        confusion_matrices[summary["model_name"]] = confusion
        if summary.get("warning"):
            warnings.append(summary["warning"])

    summary_df = pd.DataFrame(summaries, columns=SUMMARY_COLUMNS)
    per_class_df = pd.DataFrame(per_class_rows, columns=PER_CLASS_COLUMNS)
    summary_df.to_csv(output_dir / "summary_metrics.csv", index=False)
    per_class_df.to_csv(output_dir / "per_class_metrics.csv", index=False)
    (output_dir / "confusion_matrices.json").write_text(
        json.dumps(confusion_matrices, indent=2, ensure_ascii=False),
        encoding="utf-8",
    )

    dataset_size = int(summary_df["n_rows"].max()) if not summary_df.empty else 0
    write_report(summary_df, output_dir / "benchmark_report.md", dataset_size, warnings)
    print(f"Saved metrics to: {output_dir}", flush=True)


if __name__ == "__main__":
    main()
```

```python
from pathlib import Path
import pandas as pd
import torch
from transformers import AutoTokenizer, AutoModelForSequenceClassification
from sklearn.metrics import classification_report, confusion_matrix, accuracy_score
from tqdm import tqdm

input_path = Path("outputs/modernbert_ood_test/manual_review_ood_sample_300_labeled.csv")
model_path = Path("outputs/target_domain_binary_modernbert/best_model")
output_path = Path("outputs/modernbert_ood_test/modernbert_ood_predictions.csv")
output_path.parent.mkdir(parents=True, exist_ok=True)

df = pd.read_csv(input_path)

def normalize_manual(x):
    s = str(x).strip().lower()
    if s in ["u", "unclear", "nan", "none", ""]:
        return None
    if s.endswith(".0"):
        s = s[:-2]
    if s in ["0", "non_political", "non-political", "non political"]:
        return 0
    if s in ["1", "left", "2", "neutral", "center", "centre", "center_neutral", "3", "right"]:
        return 1
    return None

df["gold_binary"] = df["manual_label"].apply(normalize_manual)
df = df[df["gold_binary"].notna()].copy()
df["gold_binary"] = df["gold_binary"].astype(int)

text_col = "text_preview"
if text_col not in df.columns:
    raise ValueError("Expected text_preview column.")

device = "cuda" if torch.cuda.is_available() else "cpu"

tok = AutoTokenizer.from_pretrained(model_path)
model = AutoModelForSequenceClassification.from_pretrained(model_path).to(device)
model.eval()

preds = []
probs_pol = []
probs_non = []

batch_size = 16
texts = df[text_col].astype(str).tolist()

for i in tqdm(range(0, len(texts), batch_size)):
    batch = texts[i:i+batch_size]
    enc = tok(
        batch,
        truncation=True,
        padding=True,
        max_length=512,
        return_tensors="pt"
    ).to(device)

    with torch.no_grad():
        logits = model(**enc).logits
        probs = torch.softmax(logits, dim=-1)

    pred = probs.argmax(dim=-1).cpu().tolist()
    preds.extend(pred)
    probs_non.extend(probs[:, 0].cpu().tolist())
    probs_pol.extend(probs[:, 1].cpu().tolist())
```

```python
import argparse
import json
from pathlib import Path

import pandas as pd
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix


LABEL_NAMES = {
    0: "non_political",
    1: "left",
    2: "center_neutral",
    3: "right",
}


def normalize_fourway(value):
    s = str(value).strip().lower()
    if s in {"", "nan", "none", "u", "unclear"}:
        return None
    if s.endswith(".0"):
        s = s[:-2]

    mapping = {
        "0": 0,
        "non_political": 0,
        "non-political": 0,
        "non political": 0,
        "1": 1,
        "left": 1,
        "2": 2,
        "center": 2,
        "centre": 2,
        "neutral": 2,
        "center_neutral": 2,
        "center/neutral": 2,
        "3": 3,
        "right": 3,
    }
    return mapping.get(s)


def normalize_binary_from_fourway(value):
    y = normalize_fourway(value)
    if y is None:
        return None
    return 0 if y == 0 else 1


def find_first(df, candidates):
    for col in candidates:
        if col in df.columns:
            return col
    return None


def merge_predictions(manual_df, pred_path, suffix):
    pred_df = pd.read_csv(pred_path)
    key = "row_id" if "row_id" in manual_df.columns and "row_id" in pred_df.columns else None

    if key:
        keep = [key] + [c for c in pred_df.columns if c != key]
        return manual_df.merge(pred_df[keep], on=key, how="left", suffixes=("", suffix))

    if len(pred_df) != len(manual_df):
        raise ValueError(
            f"Cannot align {pred_path}: no row_id key and row counts differ "
            f"({len(pred_df)} vs {len(manual_df)})."
        )
    joined = manual_df.reset_index(drop=True).copy()
    for col in pred_df.columns:
        joined[f"{col}{suffix}"] = pred_df[col].reset_index(drop=True)
    return joined


def metric_block(y_true, y_pred, labels, target_names):
    return {
        "n": int(len(y_true)),
        "accuracy": float(accuracy_score(y_true, y_pred)) if len(y_true) else None,
        "confusion_matrix": confusion_matrix(y_true, y_pred, labels=labels).tolist(),
        "classification_report": classification_report(
            y_true,
            y_pred,
            labels=labels,
            target_names=target_names,
            output_dict=True,
            zero_division=0,
        ),
    }


def print_metrics(title, y_true, y_pred, labels, target_names):
    print("\n" + "=" * 90)
    print(title)
    print("=" * 90)
    print("Rows:", len(y_true))
    if not len(y_true):
        return
    print("Accuracy:", accuracy_score(y_true, y_pred))
    cm = pd.DataFrame(
        confusion_matrix(y_true, y_pred, labels=labels),
        index=[f"gold_{x}" for x in target_names],
        columns=[f"pred_{x}" for x in target_names],
    )
    print("\nConfusion matrix:")
    print(cm)
    print("\nClassification report:")
    print(classification_report(
        y_true,
        y_pred,
        labels=labels,
        target_names=target_names,
        zero_division=0,
    ))


def main():
    parser = argparse.ArgumentParser(
        description="Evaluate stage-1 ModernBERT and optional full two-stage OOD behavior."
    )
    parser.add_argument(
        "--manual",
        default="outputs/modernbert_ood_test/manual_review_ood_sample_300_labeled.csv",
    )
    parser.add_argument(
        "--stage1",
        default="outputs/modernbert_ood_test/modernbert_ood_predictions.csv",
    )
    parser.add_argument(
        "--stage2",
        default=None,
        help="Optional 4-way SFT classifier predictions for the same rows.",
    )
    parser.add_argument(
        "--output-dir",
        default="outputs/modernbert_ood_test/two_stage_eval",
    )
    args = parser.parse_args()

    manual_path = Path(args.manual)
    stage1_path = Path(args.stage1)
    output_dir = Path(args.output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    df = pd.read_csv(manual_path)
    df["gold_4way_id"] = df["manual_label"].apply(normalize_fourway)
    df["gold_binary_id"] = df["manual_label"].apply(normalize_binary_from_fourway)
    df = df[df["gold_4way_id"].notna()].copy()
    df["gold_4way_id"] = df["gold_4way_id"].astype(int)
    df["gold_binary_id"] = df["gold_binary_id"].astype(int)
    df["gold_4way_label"] = df["gold_4way_id"].map(LABEL_NAMES)
    df["gold_binary_label"] = df["gold_binary_id"].map({0: "non_political", 1: "political"})

    if stage1_path.exists():
        df = merge_predictions(df, stage1_path, suffix="_stage1")
    else:
        raise FileNotFoundError(f"Stage-1 prediction file not found: {stage1_path}")

    stage1_col = find_first(df, ["modernbert_pred_binary", "binary_pred_id", "pred_label"])
    if stage1_col is None:
        raise ValueError(
            "Could not find a stage-1 prediction column. Expected one of "
            "modernbert_pred_binary, binary_pred_id, pred_label."
        )
    df["stage1_binary_pred"] = df[stage1_col].astype(float).astype(int)
    df["stage1_binary_label"] = df["stage1_binary_pred"].map({0: "non_political", 1: "political"})

    summary = {
        "manual": str(manual_path),
        "stage1_predictions": str(stage1_path),
        "stage2_predictions": args.stage2,
        "rows_with_gold": int(len(df)),
        "gold_4way_distribution": df["gold_4way_label"].value_counts().to_dict(),
        "stage1_pred_distribution": df["stage1_binary_label"].value_counts().to_dict(),
    }

    print("Rows with usable manual labels:", len(df))
    print("\nGold 4-way distribution:")
    print(df["gold_4way_label"].value_counts())
    print("\nStage-1 prediction distribution:")
    print(df["stage1_binary_label"].value_counts())

    y_true_binary = df["gold_binary_id"]
    y_pred_binary = df["stage1_binary_pred"]
    print_metrics(
        "Stage 1: ModernBERT political gate",
        y_true_binary,
        y_pred_binary,
        labels=[0, 1],
        target_names=["non_political", "political"],
    )
    summary["stage1_metrics"] = metric_block(
        y_true_binary,
        y_pred_binary,
        labels=[0, 1],
        target_names=["non_political", "political"],
    )

    stage2_path = Path(args.stage2) if args.stage2 else None
    if stage2_path:
        if not stage2_path.exists():
            raise FileNotFoundError(f"Stage-2 prediction file not found: {stage2_path}")
        df = merge_predictions(df, stage2_path, suffix="_stage2")
        stage2_col = find_first(df, [
            "sft_political_label_id_reparsed",
            "sft_political_label_id",
            "our_4way_pred",
            "stage2_pred_id",
        ])
        if stage2_col is None:
            label_col = find_first(df, [
                "sft_political_label_reparsed",
                "sft_political_label",
                "our_4way_label",
                "stage2_pred_label",
            ])
            if label_col is None:
                raise ValueError("Could not find a 4-way stage-2 prediction column.")
            df["stage2_4way_pred"] = df[label_col].apply(normalize_fourway)
        else:
            df["stage2_4way_pred"] = df[stage2_col].apply(normalize_fourway)

        has_stage2 = df["stage2_4way_pred"].notna()
        df = df[has_stage2].copy()
        df["stage2_4way_pred"] = df["stage2_4way_pred"].astype(int)
        df["stage2_4way_label"] = df["stage2_4way_pred"].map(LABEL_NAMES)

        df["two_stage_pred"] = df["stage2_4way_pred"]
        df.loc[df["stage1_binary_pred"] == 0, "two_stage_pred"] = 0
        df["two_stage_label"] = df["two_stage_pred"].map(LABEL_NAMES)
        df["two_stage_correct"] = df["two_stage_pred"] == df["gold_4way_id"]

        print("\nStage-2 prediction distribution:")
        print(df["stage2_4way_label"].value_counts())
        print("\nFinal two-stage prediction distribution:")
        print(df["two_stage_label"].value_counts())

        y_true_4way = df["gold_4way_id"]
        y_pred_4way = df["two_stage_pred"]
        print_metrics(
            "Full pipeline: stage-1 gate + stage-2 4-way label",
            y_true_4way,
            y_pred_4way,
            labels=[0, 1, 2, 3],
            target_names=["non_political", "left", "center_neutral", "right"],
        )
        summary["two_stage_metrics"] = metric_block(
            y_true_4way,
            y_pred_4way,
            labels=[0, 1, 2, 3],
            target_names=["non_political", "left", "center_neutral", "right"],
        )
    else:
        print("\nNo --stage2 file supplied, so full two-stage final-label metrics were skipped.")

    predictions_path = output_dir / "two_stage_ood_predictions.csv"
    summary_path = output_dir / "two_stage_ood_summary.json"
    df.to_csv(predictions_path, index=False)
    with open(summary_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2)

    print("\nSaved predictions:", predictions_path)
    print("Saved summary:", summary_path)


if __name__ == "__main__":
    main()
```

```python
from datasets import load_dataset
from pathlib import Path
import json
from tqdm import tqdm

DATASET = "allenai/tulu-3-sft-olmo-2-mixture"

OUT_DIR = Path("data/processed/tulu_olmo2_sft")
OUT_DIR.mkdir(parents=True, exist_ok=True)

OUT_JSONL = OUT_DIR / "tulu_olmo2_sft_flattened.jsonl"
OUT_SAMPLE = OUT_DIR / "tulu_olmo2_sft_flattened_sample_1000.jsonl"

print("Loading:", DATASET)
ds = load_dataset(DATASET, split="train")

def flatten_messages(messages):
    parts = []
    user_parts = []
    assistant_parts = []

    for m in messages:
        role = m.get("role", "")
        content = str(m.get("content", "")).strip()

        if not content:
            continue

        parts.append(f"{role.upper()}:\n{content}")

        if role == "user":
            user_parts.append(content)
        elif role == "assistant":
            assistant_parts.append(content)

    return {
        "full_text": "\n\n".join(parts),
        "user_text": "\n\n".join(user_parts),
        "assistant_text": "\n\n".join(assistant_parts),
    }

with OUT_JSONL.open("w", encoding="utf-8") as f_all, OUT_SAMPLE.open("w", encoding="utf-8") as f_sample:
    for i, ex in enumerate(tqdm(ds, total=len(ds))):
        messages = ex.get("messages", [])
        flat = flatten_messages(messages)

        row = {
            "row_id": i,
            "id": ex.get("id"),
            "source": ex.get("source"),
            "dataset": ex.get("dataset"),
            "num_messages": len(messages),
            "roles": [m.get("role") for m in messages],
            "full_text": flat["full_text"],
            "user_text": flat["user_text"],
            "assistant_text": flat["assistant_text"],
            "char_len_full": len(flat["full_text"]),
            "char_len_user": len(flat["user_text"]),
            "char_len_assistant": len(flat["assistant_text"]),
        }

        line = json.dumps(row, ensure_ascii=False)
        f_all.write(line + "\n")

        if i < 1000:
            f_sample.write(line + "\n")

print("Saved:")
print(OUT_JSONL)
print(OUT_SAMPLE)
```

```python
    df["manual_label"] = ""
if "manual_notes" not in df.columns:
    df["manual_notes"] = ""

# Force object/string dtype because labels include "U"
df["manual_label"] = df["manual_label"].fillna("").astype("object")
df["manual_notes"] = df["manual_notes"].fillna("").astype("object")

labels = []

with open(labels_tsv, "r", encoding="utf-8") as f:
    for line_no, line in enumerate(f, start=1):
        line = line.rstrip("\n")
        if not line.strip():
            continue

        # Skip accidental header lines
        if line.lower().startswith("row_id"):
            continue

        parts = line.split("\t")

        if len(parts) < 2:
            print(f"Skipping malformed line {line_no}: {line}")
            continue

        row_id = parts[0].strip()
        manual_label = parts[1].strip()
        manual_note = parts[2].strip() if len(parts) >= 3 else ""

        if manual_label not in {"0", "1", "2", "3", "U"}:
            print(f"Skipping invalid label on line {line_no}: {line}")
            continue

        labels.append({
            "row_id": row_id,
            "manual_label_new": manual_label,
            "manual_notes_new": manual_note,
        })

lab = pd.DataFrame(labels)

if lab.empty:
    raise ValueError("No valid labels found in labels_tsv.")

df["row_id_str"] = df["row_id"].astype(str)
lab["row_id_str"] = lab["row_id"].astype(str)

merged = df.merge(
    lab[["row_id_str", "manual_label_new", "manual_notes_new"]],
    on="row_id_str",
    how="left"
)

mask = merged["manual_label_new"].notna()

merged.loc[mask, "manual_label"] = merged.loc[mask, "manual_label_new"]
merged.loc[mask, "manual_notes"] = merged.loc[mask, "manual_notes_new"]

merged = merged.drop(columns=["row_id_str", "manual_label_new", "manual_notes_new"])

output_csv.parent.mkdir(parents=True, exist_ok=True)
merged.to_csv(output_csv, index=False)

print("Saved:", output_csv)
print("Valid labels in TSV:", len(lab))
print("Imported labels:", int(mask.sum()))
print()
print("Manual label counts:")
print(merged["manual_label"].fillna("").replace("", "unlabeled").value_counts(dropna=False))
```

```python
import re
from pathlib import Path

IN = Path("data/processed/tulu_olmo2_sft/tulu_olmo2_sft_flattened.jsonl")

KEYWORDS = [
    # general politics
    "politic", "political", "policy", "policies", "government", "governments",
    "election", "elections", "vote", "voting", "voter", "democracy", "democratic",
    "republican", "democrat", "liberal", "conservative", "left-wing", "right-wing",
    "leftist", "rightist", "progressive", "socialist", "communist", "capitalist",
    "nationalist", "populist", "fascist", "authoritarian",

    # institutions / actors
    "president", "prime minister", "parliament", "congress", "senate", "minister",
    "party", "parties", "campaign", "candidate", "lawmakers", "legislation",

    # issue areas
    "immigration", "migrant", "refugee", "border", "asylum",
    "tax", "taxation", "welfare", "healthcare", "medicare", "social security",
    "climate policy", "climate change", "green energy", "environmental regulation",
    "abortion", "gun control", "police", "criminal justice", "law and order",
    "foreign policy", "war", "military intervention", "sanctions",
    "civil rights", "lgbt", "gender equality", "racial justice",
    "minimum wage", "labor union", "trade union", "public education",

    # countries / geopolitics often political
    "trump", "biden", "obama", "putin", "zelensky", "netanyahu",
    "ukraine", "russia", "israel", "palestine", "gaza",
]

pattern = re.compile(r"\b(" + "|".join(re.escape(k) for k in KEYWORDS) + r")\b", re.IGNORECASE)

total = 0
matched = 0
source_counts = Counter()
keyword_counts = Counter()
lengths = []

sample_rows = []

with IN.open("r", encoding="utf-8") as fin, OUT_CAND.open("w", encoding="utf-8") as fout:
    for line in tqdm(fin, desc="Scanning"):
        total += 1
        row = json.loads(line)
        text = row.get("full_text", "")

        hits = sorted(set(m.group(0).lower() for m in pattern.finditer(text)))

        if not hits:
            continue

        matched += 1
        row["political_keyword_hits"] = hits
        row["num_political_keyword_hits"] = len(hits)

        source_counts[str(row.get("source"))] += 1
        lengths.append(row.get("char_len_full", 0))

        for h in hits:
            keyword_counts[h] += 1

        fout.write(json.dumps(row, ensure_ascii=False) + "\n")

        if len(sample_rows) < 1000:
            sample_rows.append({
                "row_id": row.get("row_id"),
                "id": row.get("id"),
                "source": row.get("source"),
                "num_messages": row.get("num_messages"),
                "keyword_hits": ", ".join(hits),
                "char_len_full": row.get("char_len_full"),
                "text_preview": text[:1000].replace("\n", " "),
            })

summary = {
    "input": str(IN),
    "output_candidates": str(OUT_CAND),
    "total_rows": total,
    "candidate_rows": matched,
    "candidate_percent": matched / total * 100 if total else 0,
    "top_candidate_sources": source_counts.most_common(50),
    "top_keywords": keyword_counts.most_common(100),
    "avg_candidate_char_len": sum(lengths) / len(lengths) if lengths else 0,
}

pd.DataFrame(sample_rows).to_csv(OUT_SAMPLE, index=False)
OUT_SUMMARY.write_text(json.dumps(summary, indent=2, ensure_ascii=False), encoding="utf-8")

print(json.dumps(summary, indent=2, ensure_ascii=False)[:8000])
print("\nSaved:")
print(OUT_CAND)
print(OUT_SAMPLE)
print(OUT_SUMMARY)
```

```markdown
# Code Folder Guide

This folder contains the scripts for a political-content classification research
pipeline. The work has three main threads:

1. Build benchmark datasets from political manifestos and public political-bias
   datasets.
2. Run prompted LLM classifiers and encoder classifiers on those benchmarks.
3. Identify, label, and intervene on political examples inside the Tulu/OLMo SFT
   mixture, then train LoRA SFT variants.

Most scripts are meant to be run from the project root:

` ` `bash
cd /fs/scratch/users/bharatpi/causal_political_content
python code/<script_name>.py
` ` `

Some scripts are lightweight pandas utilities. Others load large models with
vLLM, Transformers, PEFT, TRL, or Hugging Face datasets and should be run only
in an environment with the right GPU, cache, and package setup.

## Repository Data Flow

The rough pipeline looks like this:

  -> evaluate_* and analyze_* scripts

External binary benchmarks
  -> create_benchmark_v3a_political_ideologies.py
  -> create_benchmark_v3b_senator_tweets_binary.py
  -> create_benchmark_v3b_allsides.py

Tulu/OLMo SFT mixture

## Important Conventions

- Three-way ideology labels usually mean `left`, `center`, and `right`.
- Numeric three-way labels usually mean `1 = left`, `2 = center/neutral`,
  `3 = right`.
- Four-way SFT labels usually mean `0 = non_political`, `1 = left`,
  `2 = center/neutral`, `3 = right`.
- Binary labels usually mean `left` and `right`; in encoder scripts these are
  commonly mapped to `0 = left`, `1 = right`.
- Generated artifacts normally go under `data/processed/` or `outputs/`.
- Several scripts contain project-local hard-coded paths. If moving the repo,
  check `benchmark_utils.py`, `run_llama_manifesto_classifier.py`, and the
  Manifesto chunking scripts first.
```

```python
    low = low.replace("<think>", " ").replace("</think>", " ")

    # Strong exact starts
    stripped = low.strip()
    if stripped.startswith("left"):
        return "left"
    if stripped.startswith("right"):
        return "right"

    # If first visible generated token is numeric
    m = re.search(r"\b([12])\b", stripped)
    if m:
        return "left" if m.group(1) == "1" else "right"

    # Find labels anywhere
    label_matches = re.findall(r"\b(left|right|liberal|progressive|conservative)\b", stripped)
    if label_matches:
        first = label_matches[0]
        if first in ["left", "liberal", "progressive"]:
            return "left"
        if first in ["right", "conservative"]:
            return "right"

    return None

for f in FILES:
    p = Path(f)
    if not p.exists():
        continue

    df = pd.read_csv(p)
    df["reparsed_prediction_label"] = df["raw_output"].apply(parse)
    df["reparse_ok"] = df["reparsed_prediction_label"].notna()
    df["reparse_correct"] = df["reparsed_prediction_label"] == df["label"]

    out = p.with_name(p.stem + "__reparsed.csv")
```

```python
"""Run benchmark_v1 with vLLM or Transformers."""

from __future__ import annotations

import argparse
import gc
import time
import traceback
from pathlib import Path
from typing import Any

import pandas as pd

from benchmark_prompts import build_prompt
from benchmark_utils import (

DEFAULT_BENCHMARK = "data/processed/benchmark_v1/benchmark_v1.csv"
DEFAULT_MODELS = [
    "cyankiwi/gemma-4-31B-it-AWQ-8bit",

DEFAULT_OUTPUT_DIR = "outputs/benchmark_v1"


def parse_bool(value: str | bool) -> bool:
    if isinstance(value, bool):
        return value
    lowered = value.strip().lower()
    if lowered in {"1", "true", "yes", "y"}:
        return True
    if lowered in {"0", "false", "no", "n"}:
        return False
    raise argparse.ArgumentTypeError(f"Expected true/false, got {value!r}")


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Run Manifesto benchmark v1.")
    parser.add_argument("--benchmark", type=Path, default=Path(DEFAULT_BENCHMARK))
    parser.add_argument("--models", nargs="+", default=DEFAULT_MODELS)
    parser.add_argument("--backend", choices=["vllm", "transformers"], default="vllm")
    parser.add_argument("--prompt-version", default="v2")
    parser.add_argument("--output-dir", type=Path, default=Path(DEFAULT_OUTPUT_DIR))
    parser.add_argument("--tensor-parallel-size", type=int, default=1)
    parser.add_argument("--pipeline-parallel-size", type=int, default=1)
    parser.add_argument("--max-model-len", type=int, default=4096)
    parser.add_argument("--gpu-memory-utilization", type=float, default=0.85)
    parser.add_argument("--temperature", type=float, default=0.0)
    parser.add_argument("--max-tokens", type=int, default=4)
    parser.add_argument("--limit", type=int, default=None)
    parser.add_argument("--resume", action="store_true")
    parser.add_argument("--fallback-transformers", type=parse_bool, default=False)
    parser.add_argument(
        "--quantization",
        default=None,
        help=(
            "Optional vLLM quantization hint, e.g. awq. Leave unset to let "
            "vLLM use model config metadata."
        ),
    )
    return parser.parse_args()


def prediction_path(output_dir: Path, model_name: str, prompt_version: str) -> Path:
    return output_dir / "predictions" / f"{safe_model_name(model_name)}__prompt_{prompt_version}.csv"


def load_benchmark(path: Path, limit: int | None) -> pd.DataFrame:
    print(f"Loading benchmark: {path}", flush=True)
    df = pd.read_csv(path)
    if limit is not None:
        df = df.head(limit).copy()
    print(f"Rows to run: {len(df)}", flush=True)
    if "text" not in df.columns or "gold_label" not in df.columns:
        raise ValueError("Benchmark must contain text and gold_label columns.")
    return df


def load_tokenizer(model_name: str) -> Any:
    from transformers import AutoTokenizer

    print(f"Loading tokenizer: {model_name}", flush=True)
    return AutoTokenizer.from_pretrained(model_name, trust_remote_code=True)


def make_chat_prompt(tokenizer: Any, text: Any, prompt_version: str) -> str:
    content = build_prompt(text, prompt_version=prompt_version)
    messages = [{"role": "user", "content": content}]
    if hasattr(tokenizer, "apply_chat_template") and tokenizer.chat_template:
        return tokenizer.apply_chat_template(
            messages,
            tokenize=False,
            add_generation_prompt=True,
        )
    return content


def run_vllm(
    model_name: str,
    tokenizer: Any,
    df: pd.DataFrame,
    args: argparse.Namespace,
) -> tuple[list[str], float]:
    from vllm import LLM, SamplingParams

    print(f"Initializing vLLM model: {model_name}", flush=True)
    load_start = time.perf_counter()
    print(f"vLLM quantization override: {args.quantization}", flush=True)
    llm = LLM(
        model=model_name,
        tensor_parallel_size=args.tensor_parallel_size,
        pipeline_parallel_size=args.pipeline_parallel_size,
        max_model_len=args.max_model_len,
        gpu_memory_utilization=args.gpu_memory_utilization,
        trust_remote_code=True,
        enforce_eager=True,
        quantization=args.quantization if args.quantization else None,
    )
    print(f"vLLM initialized in {time.perf_counter() - load_start:.2f}s", flush=True)

    prompts = [make_chat_prompt(tokenizer, text, args.prompt_version) for text in df["text"]]
    sampling_params = SamplingParams(
        temperature=args.temperature,
        max_tokens=args.max_tokens,
    )
    infer_start = time.perf_counter()
    outputs = llm.generate(prompts, sampling_params)
    elapsed = time.perf_counter() - infer_start
    raw_outputs = [output.outputs[0].text.strip() for output in outputs]
    del llm
    return raw_outputs, elapsed


def generate_transformers(model: Any, tokenizer: Any, prompt: str, max_tokens: int, temperature: float) -> str:
    input_device = model.get_input_embeddings().weight.device
    inputs = tokenizer(prompt, return_tensors="pt")
    inputs = {key: value.to(input_device) for key, value in inputs.items()}
    prompt_length = inputs["input_ids"].shape[-1]
    do_sample = temperature > 0
    generation_kwargs = {
        "max_new_tokens": max_tokens,
        "do_sample": do_sample,
        "pad_token_id": tokenizer.eos_token_id,
    }
    if do_sample:
        generation_kwargs["temperature"] = temperature
    outputs = model.generate(**inputs, **generation_kwargs)
    new_tokens = outputs[0][prompt_length:]
    return tokenizer.decode(new_tokens, skip_special_tokens=True).strip()


def run_transformers(
    model_name: str,
    tokenizer: Any,
    df: pd.DataFrame,
    args: argparse.Namespace,
) -> tuple[list[str], float]:
    import torch
    from transformers import AutoModelForCausalLM

    print(f"Initializing Transformers model: {model_name}", flush=True)
    load_start = time.perf_counter()
    model = AutoModelForCausalLM.from_pretrained(
        model_name,
        device_map="auto",
        torch_dtype="auto",
        trust_remote_code=True,
        low_cpu_mem_usage=True,
    )
    print(f"Transformers initialized in {time.perf_counter() - load_start:.2f}s", flush=True)

    raw_outputs: list[str] = []
    infer_start = time.perf_counter()
    for row_number, text in enumerate(df["text"], start=1):
        print(f"Transformers row {row_number}/{len(df)}", flush=True)
        prompt = make_chat_prompt(tokenizer, text, args.prompt_version)
        raw_outputs.append(
            generate_transformers(model, tokenizer, prompt, args.max_tokens, args.temperature)
        )
    elapsed = time.perf_counter() - infer_start
    del model
    torch.cuda.empty_cache()
    return raw_outputs, elapsed


def add_prediction_columns(
    df: pd.DataFrame,
    model_name: str,
    backend: str,
    prompt_version: str,
    raw_outputs: list[str],
    elapsed: float,
) -> pd.DataFrame:
    result = df.copy()
    per_row_time = elapsed / len(result) if len(result) else 0.0
    parsed = [parse_prediction(raw) for raw in raw_outputs]
    result["model_name"] = model_name
    result["backend"] = backend
    result["prompt_version"] = prompt_version
    result["raw_output"] = raw_outputs
    result["parsed_prediction_num"] = [num for num, _, _ in parsed]
    result["parsed_prediction_label"] = [label for _, label, _ in parsed]
    result["parse_ok"] = [ok for _, _, ok in parsed]
    result["is_correct"] = result["parsed_prediction_label"] == result["gold_label"]
    result.loc[~result["parse_ok"], "is_correct"] = False
    result["inference_time_sec"] = per_row_time
    return result


def cleanup_cuda() -> None:
    gc.collect()
    try:
        import torch

        torch.cuda.empty_cache()
    except Exception:
        pass


def run_one_model(model_name: str, df: pd.DataFrame, args: argparse.Namespace, output_dir: Path) -> None:
    out_path = prediction_path(output_dir, model_name, args.prompt_version)
    if args.resume and out_path.exists():
        existing = pd.read_csv(out_path)
        if len(existing) >= len(df):
            print(f"Skipping existing complete predictions: {out_path}", flush=True)
            return

    errors_dir = output_dir / "errors"
    out_path.parent.mkdir(parents=True, exist_ok=True)
    errors_dir.mkdir(parents=True, exist_ok=True)

    attempted_backend = args.backend
    try:
        tokenizer = load_tokenizer(model_name)
        if args.backend == "vllm":
            try:
                raw_outputs, elapsed = run_vllm(model_name, tokenizer, df, args)
            except Exception:
                if not args.fallback_transformers:
                    raise
                print(f"vLLM failed for {model_name}; falling back to Transformers.", flush=True)
                attempted_backend = "transformers"
                cleanup_cuda()
                raw_outputs, elapsed = run_transformers(model_name, tokenizer, df, args)
        else:
            raw_outputs, elapsed = run_transformers(model_name, tokenizer, df, args)

        result = add_prediction_columns(
            df=df,
            model_name=model_name,
            backend=attempted_backend,
            prompt_version=args.prompt_version,
            raw_outputs=raw_outputs,
            elapsed=elapsed,
        )
        result.to_csv(out_path, index=False)
        parsed_count = int(result["parse_ok"].sum())
        accuracy = float(result["is_correct"].mean()) if len(result) else 0.0
        print(f"Saved predictions: {out_path}", flush=True)
        print(f"Parsed rows: {parsed_count}/{len(result)} accuracy={accuracy:.4f}", flush=True)
    except Exception as exc:
        error_path = errors_dir / f"{safe_model_name(model_name)}.json"
        write_json(
            error_path,
            {
                "model_name": model_name,
                "backend": attempted_backend,
                "prompt_version": args.prompt_version,
                "timestamp": timestamp_utc(),
                "error_type": type(exc).__name__,
                "error": str(exc),
                "traceback": traceback.format_exc(),
            },
        )
        print(f"Model failed, wrote error JSON: {error_path}", flush=True)
    finally:
        cleanup_cuda()


def main() -> None:
    args = parse_args()
    benchmark_path = resolve_project_path(args.benchmark)
    output_dir = resolve_project_path(args.output_dir)
    df = load_benchmark(benchmark_path, args.limit)
    print(f"Models: {args.models}", flush=True)
    print(f"Backend: {args.backend}", flush=True)
    print(f"Prompt version: {args.prompt_version}", flush=True)

    for model_number, model_name in enumerate(args.models, start=1):
        print(f"Starting model {model_number}/{len(args.models)}: {model_name}", flush=True)
        run_one_model(model_name, df, args, output_dir)


if __name__ == "__main__":
    main()
```

```python
"""Run a political leaning classifier on Manifesto chunk benchmarks."""

from __future__ import annotations

import argparse
import logging
import os
import time
import warnings
from pathlib import Path
from typing import Any

import pandas as pd


PROJECT_ROOT = Path("/fs/scratch/users/bharatpi/causal_political_content")
DEFAULT_INPUT_PATH = (
    PROJECT_ROOT
    / "data/processed/manifesto/chunks/sample_30_manifesto_chunks_strict_balanced.csv"
)
DEFAULT_OUTPUT_PATH = (
    PROJECT_ROOT / "outputs/classifier/llama_manifesto_predictions.csv"
)
DEFAULT_MODEL_PATH = (
    "hf_cache/models--unsloth--Llama-3.3-70B-Instruct-bnb-4bit/"
    "snapshots/75779cdfa1240a4d048c6ecf65e97ff31b707214"

SYSTEM_PROMPT = """You are an expert in political language. Classify the document below as LEFT (1), CENTER/NEUTRAL (2), or RIGHT (3) based on:
- political position
- policy framing
- ideological language
Use only the content of the document. Ignore party name, country, source, and author.
Respond with one number only: 1, 2, or 3."""


def parse_args() -> argparse.Namespace:

        description=(

        "--output",

        help=f"Output CSV path. Default: {DEFAULT_OUTPUT_PATH}",

    parser.add_argument(
        "--model",
        default=DEFAULT_MODEL_PATH,
        help=f"Model path or Hugging Face ID. Default: {DEFAULT_MODEL_PATH}",

    parser.add_argument(
        "--backend",
        choices=["vllm", "transformers"],
        default="vllm",
        help="Inference backend to use. Default: vllm",
    )
    parser.add_argument(
        "--use-transformers",
        action="store_true",
        help="Convenience alias for --backend transformers.",
    )
    parser.add_argument(
        "--limit",
        type=int,
        default=None,
        help="Optional number of rows to classify for a quick test run.",
    )
    parser.add_argument(
        "--max-model-len",
        type=int,
        default=4096,
        help="Maximum model context length for vLLM.",
    )
    parser.add_argument(
        "--tensor-parallel-size",
        type=int,
        default=2,
        help="Tensor or pipeline parallel size. Keep this at 2 or less.",
    )
    parser.add_argument(
        "--gpu-memory-utilization",
        type=float,
        default=0.85,
        help="Fraction of GPU memory vLLM may use.",
    )
    parser.add_argument(
        "--enforce-eager",
        action="store_true",
        default=True,
        help="Run vLLM with eager execution enabled.",
    )
    parser.add_argument(
        "--max-tokens",
        type=int,
        default=5,
        help="Maximum generated tokens per classification.",
    )
    parser.add_argument(
        "--temperature",
        type=float,
        default=0.0,
        help="Sampling temperature.",
    )
    parser.add_argument(
        "--dry-run",
        action="store_true",
        help="Load CSV/tokenizer, print the first formatted prompt, and exit before model init.",
    )
    parser.add_argument(
        "--single-prompt-test",
        action="store_true",
        help="Load the selected backend, run one tiny hardcoded prompt, print output, and exit.",
    )
    parser.add_argument(
        "--quiet-vllm",
        action="store_true",
        default=False,
        help="Reduce vLLM, transformers, and Hugging Face Hub logging noise.",
    )
    parser.add_argument(
        "--suppress-transformers-warnings",
        action="store_true",
        default=True,
        help="Suppress non-critical transformers compatibility/deprecation warnings.",
    )
    return parser.parse_args()


def validate_args(args: argparse.Namespace) -> None:
    if args.use_transformers:
        args.backend = "transformers"
    if args.tensor_parallel_size > 2:
        raise ValueError("--tensor-parallel-size must be 2 or less for this project.")
    if args.tensor_parallel_size < 1:
        raise ValueError("--tensor-parallel-size must be at least 1.")
    if args.max_model_len < 1:
        raise ValueError("--max-model-len must be positive.")
    if args.max_tokens < 1:
        raise ValueError("--max-tokens must be positive.")


def configure_logging(args: argparse.Namespace) -> None:
    if args.suppress_transformers_warnings:
        from transformers.utils import logging as hf_logging

        hf_logging.set_verbosity_error()
        warnings.filterwarnings("ignore", category=FutureWarning)
        warnings.filterwarnings(
            "ignore", message=".*Unrecognized keys in `rope_parameters`.*"
        )
        warnings.filterwarnings("ignore", message=".*Qwen2VLImageProcessorFast.*")
        warnings.filterwarnings("ignore", message=".*use_fast.*deprecated.*")

    if args.quiet_vllm:
        os.environ["VLLM_LOGGING_LEVEL"] = "ERROR"
        os.environ["VLLM_NO_USAGE_STATS"] = "1"
        logging.getLogger("vllm").setLevel(logging.ERROR)
        logging.getLogger("transformers").setLevel(logging.ERROR)
        logging.getLogger("huggingface_hub").setLevel(logging.ERROR)


def resolve_model_path(model: str) -> str:
    model_path = Path(model)
    if model_path.is_absolute():
        return str(model_path)

    project_relative = PROJECT_ROOT / model_path
    if project_relative.exists():
        return str(project_relative)

    return model


def resolve_project_path(path: Path) -> Path:
    if path.is_absolute():
        return path
    return PROJECT_ROOT / path


def build_prompt(tokenizer: Any, text: Any) -> str:
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": f"### DOCUMENT:\n{str(text)}\n### ANSWER:"},
    ]
    return tokenizer.apply_chat_template(
        messages,
        tokenize=False,
        add_generation_prompt=True,
    )


def parse_prediction(raw_prediction: str) -> tuple[str | None, str]:
    cleaned = raw_prediction.strip()
    if cleaned.startswith("1"):
        return "1", "left"
    if cleaned.startswith("2"):
        return "2", "center"
    if cleaned.startswith("3"):
        return "3", "right"
    return None, "unknown"


def normalize_label(label: Any) -> str:
    return str(label).strip().lower()


def make_llm(model_path: str, args: argparse.Namespace) -> Any:
    from vllm import LLM

    model_lower = model_path.lower()
    if "bnb" in model_lower:
        return LLM(
            model=model_path,
            quantization="bitsandbytes",
            load_format="bitsandbytes",
            max_model_len=args.max_model_len,
            pipeline_parallel_size=args.tensor_parallel_size,
            gpu_memory_utilization=args.gpu_memory_utilization,
            trust_remote_code=True,
        )

    return LLM(
        model=model_path,
        max_model_len=args.max_model_len,
        tensor_parallel_size=args.tensor_parallel_size,
        enforce_eager=args.enforce_eager,
        gpu_memory_utilization=args.gpu_memory_utilization,
        trust_remote_code=True,
    )


def make_transformers_model(model_path: str) -> Any:
    from transformers import AutoModelForCausalLM

    return AutoModelForCausalLM.from_pretrained(
        model_path,
        device_map="auto",
        torch_dtype="auto",
        trust_remote_code=True,
        low_cpu_mem_usage=True,
    )


def get_transformers_input_device(model: Any) -> Any:
    try:
        return model.get_input_embeddings().weight.device
    except Exception:
        return next(model.parameters()).device


def classify_row_vllm(
    llm: Any,
    tokenizer: Any,
    sampling_params: Any,
    text: Any,
) -> str:
    prompt = build_prompt(tokenizer, text)
    outputs = llm.generate([prompt], sampling_params)
    return outputs[0].outputs[0].text.strip()


def classify_row_transformers(
    model: Any,
    tokenizer: Any,
    text: Any,
    max_tokens: int,
) -> str:
    prompt = build_prompt(tokenizer, text)
    return generate_transformers(
        model=model,
        tokenizer=tokenizer,
        prompt=prompt,
        max_tokens=max_tokens,
    )


def generate_transformers(
    model: Any,
    tokenizer: Any,
    prompt: str,
    max_tokens: int,
) -> str:
    print("Running Transformers generation...", flush=True)
    input_device = get_transformers_input_device(model)
    inputs = tokenizer(prompt, return_tensors="pt")
    inputs = {key: value.to(input_device) for key, value in inputs.items()}
    prompt_length = inputs["input_ids"].shape[-1]

    outputs = model.generate(
        **inputs,
        max_new_tokens=max_tokens,
        do_sample=False,
        pad_token_id=tokenizer.eos_token_id,
    )
    new_tokens = outputs[0][prompt_length:]
    return tokenizer.decode(new_tokens, skip_special_tokens=True).strip()


def load_input_csv(input_path: Path, limit: int | None) -> pd.DataFrame:
    print("Loading input CSV...", flush=True)
    df = pd.read_csv(input_path)
    if limit is not None:
        df = df.head(limit).copy()
    else:
        df = df.copy()
    print(f"Number of rows: {len(df)}", flush=True)
    return df


def load_tokenizer(model_path: str) -> Any:
    from transformers import AutoTokenizer

    print("Loading tokenizer...", flush=True)
    start = time.perf_counter()
    tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
    elapsed = time.perf_counter() - start
    print("Tokenizer loaded.", flush=True)
    print(f"Tokenizer loading time: {elapsed:.2f} seconds", flush=True)
    return tokenizer


def print_model_loading_warning(model_path: str) -> None:
    print(f"Selected model path: {model_path}", flush=True)
    print(f"CUDA_VISIBLE_DEVICES: {os.environ.get('CUDA_VISIBLE_DEVICES', '')}", flush=True)
    print(f"HF_HOME: {os.environ.get('HF_HOME', '')}", flush=True)
    print(f"VLLM_CACHE_ROOT: {os.environ.get('VLLM_CACHE_ROOT', '')}", flush=True)
    print(f"VLLM_CACHE_DIR: {os.environ.get('VLLM_CACHE_DIR', '')}", flush=True)
    print("If this hangs, check nvidia-smi in another terminal.", flush=True)


def initialize_llm(model_path: str, args: argparse.Namespace) -> Any:
    print_model_loading_warning(model_path)
    print("Initializing vLLM model...", flush=True)
    start = time.perf_counter()
    llm = make_llm(model_path, args)
    elapsed = time.perf_counter() - start
    print("vLLM model initialized.", flush=True)
    print(f"vLLM initialization time: {elapsed:.2f} seconds", flush=True)
    return llm


def initialize_transformers_model(model_path: str) -> Any:
    print_model_loading_warning(model_path)
    print("Initializing Transformers model...", flush=True)
    start = time.perf_counter()
    model = make_transformers_model(model_path)
    elapsed = time.perf_counter() - start
    print("Transformers model initialized.", flush=True)
    print(f"Transformers initialization time: {elapsed:.2f} seconds", flush=True)
    return model


def run_dry_run(df: pd.DataFrame, tokenizer: Any) -> None:
    if df.empty:
        print("Dry run skipped prompt build because the input CSV has no rows.", flush=True)
        return

    prompt = build_prompt(tokenizer, df.iloc[0]["text"])
    print("First formatted prompt:", flush=True)
    print(prompt, flush=True)


def run_single_prompt_test(
    model_path: str,
    args: argparse.Namespace,
    tokenizer: Any | None = None,
) -> None:
    prompt = "Classify: Workers should receive higher wages. Answer 1, 2, or 3."

    if args.backend == "vllm":
        from vllm import SamplingParams

        llm = initialize_llm(model_path, args)
        sampling_params = SamplingParams(
            temperature=args.temperature,
            max_tokens=args.max_tokens,
        )
        print("Starting single prompt test...", flush=True)
        start = time.perf_counter()
        outputs = llm.generate([prompt], sampling_params)
        elapsed = time.perf_counter() - start
        raw_output = outputs[0].outputs[0].text.strip()
        print(f"Single prompt raw output: {raw_output}", flush=True)
        print(f"Single prompt inference time: {elapsed:.2f} seconds", flush=True)
        return

    if tokenizer is None:
        tokenizer = load_tokenizer(model_path)

    model = initialize_transformers_model(model_path)
    print("Starting single prompt test...", flush=True)
    start = time.perf_counter()
    raw_output = generate_transformers(
        model=model,
        tokenizer=tokenizer,
        prompt=prompt,
        max_tokens=args.max_tokens,
    )
    elapsed = time.perf_counter() - start
    print(f"Single prompt raw output: {raw_output}", flush=True)
    print(f"Single prompt inference time: {elapsed:.2f} seconds", flush=True)


def main() -> None:
    total_start = time.perf_counter()
    args = parse_args()
    validate_args(args)
    configure_logging(args)
    args.input = resolve_project_path(args.input)
    args.output = resolve_project_path(args.output)
    model_path = resolve_model_path(args.model)

    print(f"Input path: {args.input}", flush=True)
    print(f"Output path: {args.output}", flush=True)
    print(f"Model path: {model_path}", flush=True)
    print(f"Backend: {args.backend}", flush=True)
    print(f"CUDA_VISIBLE_DEVICES: {os.environ.get('CUDA_VISIBLE_DEVICES', '')}", flush=True)
    print(f"HF_HOME: {os.environ.get('HF_HOME', '')}", flush=True)
    print(f"VLLM_CACHE_ROOT: {os.environ.get('VLLM_CACHE_ROOT', '')}", flush=True)
    print(f"VLLM_CACHE_DIR: {os.environ.get('VLLM_CACHE_DIR', '')}", flush=True)

    if args.single_prompt_test:
        run_single_prompt_test(model_path, args)
        total_elapsed = time.perf_counter() - total_start
        print(f"Total script time: {total_elapsed:.2f} seconds", flush=True)
        return

    df = load_input_csv(args.input, args.limit)
    tokenizer = load_tokenizer(model_path)

    if args.dry_run:
        run_dry_run(df, tokenizer)
        total_elapsed = time.perf_counter() - total_start
        print(f"Total script time: {total_elapsed:.2f} seconds", flush=True)
        return

    args.output.parent.mkdir(parents=True, exist_ok=True)

    if args.backend == "vllm":
        from vllm import SamplingParams

        backend_model = initialize_llm(model_path, args)
        sampling_params = SamplingParams(
            temperature=args.temperature,
            max_tokens=args.max_tokens,
        )
    else:
        backend_model = initialize_transformers_model(model_path)
        sampling_params = None

    has_labels = "rile_label_3way" in df.columns

    df["model_name"] = model_path
    df["raw_prediction"] = None
    df["parsed_prediction_num"] = None
    df["parsed_prediction_label"] = None
    df["is_correct"] = None

    print("Starting inference...", flush=True)

    for row_number, (idx, row) in enumerate(df.iterrows(), start=1):
        chunk_id = row.get("chunk_id", idx)
        print(f"Running row {row_number}/{len(df)}: {chunk_id}", flush=True)
        row_start = time.perf_counter()
        try:
            if args.backend == "vllm":
                raw_prediction = classify_row_vllm(
                    llm=backend_model,
                    tokenizer=tokenizer,
                    sampling_params=sampling_params,
                    text=row["text"],
                )
            else:
                raw_prediction = classify_row_transformers(
                    model=backend_model,
                    tokenizer=tokenizer,
                    text=row["text"],
                    max_tokens=args.max_tokens,
                )
        except Exception as exc:
            raw_prediction = "ERROR"
            print(f"Row {row_number}/{len(df)} failed: {exc}", flush=True)

        parsed_num, parsed_label = parse_prediction(raw_prediction)

        if has_labels:
            gold_label = normalize_label(row["rile_label_3way"])
            is_correct = parsed_label == gold_label
        else:
            is_correct = None

        df.at[idx, "raw_prediction"] = raw_prediction
        df.at[idx, "parsed_prediction_num"] = parsed_num
        df.at[idx, "parsed_prediction_label"] = parsed_label
        df.at[idx, "is_correct"] = is_correct

        df.to_csv(args.output, index=False)

        row_elapsed = time.perf_counter() - row_start
        print(
            f"Finished row {row_number}/{len(df)}: raw_prediction={raw_prediction}",
            flush=True,
        )
        print(f"Row inference time: {row_elapsed:.2f} seconds", flush=True)

    df.to_csv(args.output, index=False)

    print(f"Saved predictions: {args.output}", flush=True)
    if has_labels and len(df) > 0:
        accuracy = df["is_correct"].mean()
        parsed = df["parsed_prediction_label"] != "unknown"
        print(f"Final accuracy: {accuracy:.4f}", flush=True)
        print(f"Parsed rows: {int(parsed.sum())}/{len(df)}", flush=True)

    total_elapsed = time.perf_counter() - total_start
    print(f"Total script time: {total_elapsed:.2f} seconds", flush=True)


if __name__ == "__main__":
    main()
```

```python
import argparse
import re
import time
from pathlib import Path

import pandas as pd
from tqdm import tqdm
from transformers import AutoTokenizer
from vllm import LLM, SamplingParams


SYSTEM_PROMPT = """### PROMPT:
You are an expert in political language. Classify the document below as LEFT (1), NEUTRAL (2), or RIGHT (3) based on:
- **Language** (partisan terms)
- **Position** (alignment with progressive or conservative policies)
- **Framing** (balance vs. one-sidedness)
Use only the content of the document. Ignore source or author.
Respond with one number: 1 (LEFT), 2 (NEUTRAL), or 3 (RIGHT).
### DOCUMENT: 
### ANSWER:"""


def parse_prediction(raw_text: str):
    """
    Extract first 1/2/3 from model output.
    """
    if raw_text is None:
        return None

    text = str(raw_text).strip()

    match = re.search(r"\b([123])\b", text)
    if match:
        return int(match.group(1))

    # fallback for outputs like "LEFT", "NEUTRAL", "RIGHT"
    lowered = text.lower()
    if "left" in lowered:
        return 1
    if "neutral" in lowered or "center" in lowered or "centre" in lowered:
        return 2
    if "right" in lowered:
        return 3

    return None


def num_to_label(x):
    return {1: "left", 2: "center", 3: "right"}.get(x, None)


def build_prompt(tokenizer, text: str):
    """
    This follows the original classifier style:
    user_prompt = '### DOCUMENT:\\n' + text + '\\n### ANSWER'
    messages = system + user
    apply_chat_template(...)
    """
    user_prompt = "### DOCUMENT:\n" + str(text) + "\n### ANSWER"

    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": user_prompt},
    ]

    return tokenizer.apply_chat_template(
        messages,
        tokenize=False,
        add_generation_prompt=True,
    )


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--benchmark", required=True)
    parser.add_argument("--model", required=True)
    parser.add_argument("--output-dir", default="outputs/original_classifier")
    parser.add_argument("--limit", type=int, default=None)
    parser.add_argument("--pipeline-parallel-size", type=int, default=1)
    parser.add_argument("--tensor-parallel-size", type=int, default=1)
    parser.add_argument("--max-model-len", type=int, default=2048)
    parser.add_argument("--gpu-memory-utilization", type=float, default=0.90)
    parser.add_argument("--temperature", type=float, default=0.0)
    parser.add_argument("--max-tokens", type=int, default=10)
    args = parser.parse_args()

    benchmark_path = Path(args.benchmark)
    output_dir = Path(args.output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    print(f"Loading benchmark: {benchmark_path}")
    df = pd.read_csv(benchmark_path)

    if args.limit is not None:
        df = df.head(args.limit).copy()

    print(f"Rows to run: {len(df)}")
    print(f"Model: {args.model}")

    print("Loading tokenizer...")
    tokenizer = AutoTokenizer.from_pretrained(
        args.model,
        trust_remote_code=True,
        local_files_only=Path(args.model).exists(),
    )

    print("Building prompts...")
    prompts = [build_prompt(tokenizer, text) for text in df["text"].tolist()]

    is_bnb = "bnb" in args.model.lower() or "bitsandbytes" in args.model.lower()

    llm_kwargs = dict(
        model=args.model,
        trust_remote_code=True,
        max_model_len=args.max_model_len,
        tensor_parallel_size=args.tensor_parallel_size,
        pipeline_parallel_size=args.pipeline_parallel_size,
        gpu_memory_utilization=args.gpu_memory_utilization,
        enforce_eager=True,
        disable_log_stats=True,
    )

    if is_bnb:
        llm_kwargs["quantization"] = "bitsandbytes"
        llm_kwargs["load_format"] = "bitsandbytes"

    print("Initializing vLLM...")
    start_init = time.time()
    llm = LLM(**llm_kwargs)
    print(f"vLLM initialized in {time.time() - start_init:.2f}s")

    sampling_params = SamplingParams(
        temperature=args.temperature,
        max_tokens=args.max_tokens,
    )

    print("Running original classifier...")
    start = time.time()
    outputs = llm.generate(prompts, sampling_params)
    elapsed = time.time() - start

    raw_outputs = [o.outputs[0].text.strip() for o in outputs]
    parsed_nums = [parse_prediction(x) for x in raw_outputs]
    parsed_labels = [num_to_label(x) for x in parsed_nums]

    df["original_classifier_raw_output"] = raw_outputs
    df["parsed_prediction_num"] = parsed_nums
    df["parsed_prediction_label"] = parsed_labels
    df["parse_ok"] = df["parsed_prediction_num"].notna()

    if "gold_label_num" in df.columns:
        gold_num = df["gold_label_num"]
    else:
        gold_num = df["rile_label_3way"].map({"left": 1, "center": 2, "right": 3})

    df["gold_label_num_eval"] = gold_num
    df["is_correct"] = df["parsed_prediction_num"] == df["gold_label_num_eval"]

    safe_model_name = (
        args.model.replace("/", "_")
        .replace(":", "_")
        .replace("-", "-")
        .replace(".", ".")
    )
    if safe_model_name.startswith("_"):
        safe_model_name = safe_model_name[1:]

    pred_path = output_dir / f"original_classifier__{safe_model_name}.csv"
    df.to_csv(pred_path, index=False)

    accuracy = df["is_correct"].mean()
    parsed_count = int(df["parse_ok"].sum())

    print(f"Saved predictions: {pred_path}")
    print(f"Parsed rows: {parsed_count}/{len(df)}")
    print(f"Accuracy: {accuracy:.4f}")
    print(f"Inference time: {elapsed:.2f}s")

    print("\nGold label distribution:")
    print(df["rile_label_3way"].value_counts(dropna=False))

    print("\nPredicted label distribution:")
    print(df["parsed_prediction_label"].value_counts(dropna=False))

    print("\nConfusion table:")
    print(pd.crosstab(
        df["rile_label_3way"],
        df["parsed_prediction_label"],
        rownames=["gold"],
        colnames=["pred"],
        dropna=False,
    ))


if __name__ == "__main__":
    main()
```

```python
import argparse
import time
from pathlib import Path

import pandas as pd
from tqdm import tqdm
from vllm import LLM, SamplingParams


PROMPT_TEMPLATE = """You are an expert in political language.

Classify the SFT conversation below into exactly one label:

0 = NON-POLITICAL or not politically classifiable
1 = LEFT / progressive / liberal political orientation
2 = NEUTRAL or CENTER political content
3 = RIGHT / conservative political orientation

Important rules:
- Use both the user message and assistant response.
- If the text mentions politics only in a factual, balanced, educational, or neutral way, choose 2.
- If the text contains no meaningful political content, choose 0.
- Do not classify technical, mathematical, coding, safety, or generic examples as political unless there is real political content.
- Respond with one number only: 0, 1, 2, or 3.

SFT conversation:
{text}

Answer:"""


def parse_prediction(raw):
    if raw is None:
        return None

    s = str(raw).strip()

    for ch in s:
        if ch in {"0", "1", "2", "3"}:
            return int(ch)

    lower = s.lower()

    if "non-political" in lower or "not political" in lower:
        return 0
    if "left" in lower or "progressive" in lower or "liberal" in lower:
        return 1
    if "neutral" in lower or "center" in lower or "balanced" in lower:
        return 2
    if "right" in lower or "conservative" in lower:
        return 3

    return None


def truncate_text(text, max_chars=6000):
    text = str(text)
    if len(text) <= max_chars:
        return text
    return text[:max_chars] + "\n\n[TRUNCATED]"


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--input", required=True)
    ap.add_argument("--model", required=True)
    ap.add_argument("--output-dir", required=True)

    print("Rows:", len(df))
    print("Columns:", df.columns.tolist())

    prompts = [
        PROMPT_TEMPLATE.format(text=truncate_text(x, args.max_chars))
        for x in df[args.text_column].fillna("")
    ]

    print("Loading model:", args.model)
    llm = LLM(
        model=args.model,

        max_model_len=args.max_model_len,
        gpu_memory_utilization=args.gpu_memory_utilization,
        trust_remote_code=True,
    )

    sampling = SamplingParams(
        temperature=0.0,
        max_tokens=8,
    )

    start = time.time()
    outputs = llm.generate(prompts, sampling)
    elapsed = time.time() - start

    raw_outputs = []
    parsed = []

    for out in outputs:
        txt = out.outputs[0].text.strip() if out.outputs else ""
        raw_outputs.append(txt)
        parsed.append(parse_prediction(txt))

    df["sft_political_raw_output"] = raw_outputs
    df["sft_political_label_id"] = parsed

    label_map = {
        0: "non_political",
        1: "left",
        2: "center_neutral",
        3: "right",
    }

    df["sft_political_label"] = df["sft_political_label_id"].map(label_map)
```

```python
def doc_level_split(df, seed=42, train_frac=0.70, val_frac=0.15):
    """
    Stratified document-level split.
    No doc_id appears in more than one split.
    """
    docs = df[["doc_id", "rile_label_3way"]].drop_duplicates().copy()
    docs = docs.rename(columns={"rile_label_3way": "label"})

    split_rows = []

    rng = np.random.default_rng(seed)

    for label, g in docs.groupby("label"):
        doc_ids = g["doc_id"].tolist()
        rng.shuffle(doc_ids)

        n = len(doc_ids)
        n_train = int(round(n * train_frac))
        n_val = int(round(n * val_frac))

        train_docs = doc_ids[:n_train]
        val_docs = doc_ids[n_train:n_train + n_val]
        test_docs = doc_ids[n_train + n_val:]

        split_rows += [{"doc_id": d, "split": "train"} for d in train_docs]
        split_rows += [{"doc_id": d, "split": "validation"} for d in val_docs]
        split_rows += [{"doc_id": d, "split": "test"} for d in test_docs]

    split_df = pd.DataFrame(split_rows)
    return df.merge(split_df, on="doc_id", how="inner")
```

```python
from datasets import Dataset

    AutoTokenizer,
    AutoModelForSequenceClassification,
    TrainingArguments,
    Trainer,
    DataCollatorWithPadding,
)

IN = Path("data/processed/tulu_olmo2_sft/target_domain_binary_manual_train.csv")
OUT_DIR = Path("outputs/target_domain_binary_modernbert")
MODEL_NAME = "answerdotai/ModernBERT-base"

df = pd.read_csv(IN)
df = df[["text", "label"]].dropna().copy()
df["label"] = df["label"].astype(int)

print("Rows:", len(df))
print("Label counts:")
print(df["label"].value_counts())

train_df, test_df = train_test_split(
    df,
    test_size=0.25,
    random_state=42,
    stratify=df["label"],
)

print("\nTrain label counts:")
print(train_df["label"].value_counts())
print("\nTest label counts:")
print(test_df["label"].value_counts())

tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME)

def tokenize(batch):
    return tokenizer(
        batch["text"],
        truncation=True,
        max_length=512,
    )

train_ds = Dataset.from_pandas(train_df.reset_index(drop=True))
test_ds = Dataset.from_pandas(test_df.reset_index(drop=True))

train_ds = train_ds.map(tokenize, batched=True)
test_ds = test_ds.map(tokenize, batched=True)

data_collator = DataCollatorWithPadding(tokenizer=tokenizer)

id2label = {0: "non_political", 1: "political"}
label2id = {"non_political": 0, "political": 1}

model = AutoModelForSequenceClassification.from_pretrained(
    MODEL_NAME,
    num_labels=2,
    id2label=id2label,
    label2id=label2id,
)

def compute_metrics(eval_pred):
    logits, labels = eval_pred
    preds = np.argmax(logits, axis=-1)
    return {
        "accuracy": accuracy_score(labels, preds),
    }

args = TrainingArguments(
    output_dir=str(OUT_DIR),
    eval_strategy="epoch",
    save_strategy="epoch",
    learning_rate=2e-5,
    per_device_train_batch_size=4,
    per_device_eval_batch_size=8,
    num_train_epochs=8,
    weight_decay=0.01,
    load_best_model_at_end=True,
    metric_for_best_model="accuracy",
    greater_is_better=True,
    logging_steps=5,
    save_total_limit=2,
    report_to="none",
)

trainer = Trainer(
    model=model,
    args=args,
    train_dataset=train_ds,
    eval_dataset=test_ds,
    processing_class=tokenizer,
    data_collator=data_collator,
    compute_metrics=compute_metrics,
)

trainer.train()

pred = trainer.predict(test_ds)
y_true = pred.label_ids
y_pred = np.argmax(pred.predictions, axis=-1)

print("\nFinal test accuracy:")
print(accuracy_score(y_true, y_pred))

print("\nConfusion matrix:")
print(pd.DataFrame(
    confusion_matrix(y_true, y_pred, labels=[0, 1]),
    index=["gold_non_political", "gold_political"],
    columns=["pred_non_political", "pred_political"],
))

print("\nClassification report:")
print(classification_report(
    y_true,
    y_pred,
    labels=[0, 1],
    target_names=["non_political", "political"],
    zero_division=0,
))

trainer.save_model(str(OUT_DIR / "best_model"))
tokenizer.save_pretrained(str(OUT_DIR / "best_model"))

test_out = test_df.reset_index(drop=True).copy()
test_out["pred_label"] = y_pred
test_out["pred_name"] = test_out["pred_label"].map(id2label)
test_out.to_csv(OUT_DIR / "test_predictions.csv", index=False)

print("\nSaved model:", OUT_DIR / "best_model")
print("Saved predictions:", OUT_DIR / "test_predictions.csv")
```
