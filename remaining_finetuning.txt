cat >| scripts/run_remaining_olmo3_interventions_4gpu.sh <<'BASH'
#!/usr/bin/env bash
set -euo pipefail

cd /fs/scratch/users/bharatpi/causal_political_content

PYTORCH_RUN="/fs/scratch/users/bharatpi/causal_political_content/envs/classifier/bin/torchrun"
TRAIN_SCRIPT="code/train_olmo3_lora_dolci_intervention.py"

export CUDA_VISIBLE_DEVICES=2,3,4,5
export PYTHONUNBUFFERED=1
export TOKENIZERS_PARALLELISM=false
export HF_HOME=/fs/scratch/users/bharatpi/causal_political_content/hf_cache
export NCCL_DEBUG=WARN

CONDITIONS=(
    no_pol
    lr_balanced
    no_ideology
    no_left
    no_right
)

for CONDITION in "${CONDITIONS[@]}"
do
    OUTPUT_DIR="outputs/sft_runs/olmo3_union_t04_${CONDITION}_full_dataset_4gpu"
    LOG_FILE="logs/train_olmo3_union_t04_${CONDITION}_full_dataset_4gpu.log"

    echo "===================================================================================================="
    echo "STARTING CONDITION: ${CONDITION}"
    echo "START TIME: $(date)"
    echo "OUTPUT: ${OUTPUT_DIR}"
    echo "LOG: ${LOG_FILE}"
    echo "===================================================================================================="

    "$PYTORCH_RUN" \
        --standalone \
        --nproc_per_node=4 \
        "$TRAIN_SCRIPT" \
        --condition "$CONDITION" \
        --output-dir "$OUTPUT_DIR" \
        --full-dataset-pass \
        --batch-size 1 \
        --gradient-accumulation-steps 16 \
        --learning-rate 2e-5 \
        --max-seq-length 2048 \
        --logging-steps 25 \
        --save-steps 1000 \
        --save-total-limit 3 \
        --shuffle-buffer-size 100000 \
        > "$LOG_FILE" 2>&1

    echo "FINISHED CONDITION: ${CONDITION}"
    echo "FINISH TIME: $(date)"
done

echo "ALL REMAINING CONDITIONS FINISHED: $(date)"
BASH

chmod +x scripts/run_remaining_olmo3_interventions_4gpu.sh
